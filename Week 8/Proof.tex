\section{Introduction (ND)}

Fix $G$ as a finite group. Let $P_G$ be the \emph{associativity $G$-operad} in $\Sets_G$, defined by $P_G(n):= \Sets(G,\Sigma_n)$, with structure maps defined by post composition
\begin{align*}
    \gamma:\Sets(G,\Sigma_k)\times \prod_{r=1}^k\Sets(G,\Sigma_{j_r})\cong \Sets\left(G,\Sigma_k\times \prod_{r=1}^k \Sigma_{j_r}\right)\xrightarrow{\ \gamma\circ\_\ }\Sets(G,\Sigma_j),
\end{align*},
where $j=\sum_{r=1}^k j_r$, and $\gamma:\Sigma_k\times \prod_{r=1}^k \Sigma_{j_r}\rightarrow \Sigma_j$ is the structure map for associativity operad $\mathscr{M}$ in $\Sets$. This is naturally a free $(G\times\Sigma)$-operad, as the right $(G\times \Sigma_n)$-action at each $P_G(n)$ is free.

However, it is mentioned in \cite{BBKO21} that such $G$-operad is not finitely generated, as a rough explanation, for every $n\in\NN$, there exists $m>n$ and a function $f:G\rightarrow \Sigma_m$, such that it cannot be generated using $\{P_G(k)\}_{k\leq n}$ with the operadic structure map and the $(G\times \Sigma)$-actions.

\hfil

As a substitution, \cite{BBKO21} defines a finitely-generated suboperad $Q_{G,\mathfrak{D}}\hookrightarrow P_G$ (for some parameter $\mathfrak{D}$ that's associated with choice of transitive $G$-sets), such that after chaotic generalization the two are equivalent in $\Cat_G$, and having equivalent categories of operadic algebras.

We'll mimic the statement, define an intermediate operad $Q_{G,\mathfrak{D}}\hookrightarrow Q_G\hookrightarrow P_G$, promote it to $\Q_G$ in $\Cat_G$, and \emph{attempt to} modify the proof in \cite{LMRZ26} to show the equivalence of 2-categories between $\Q_G$-$\PsAlg$ and $\F_{G,\RR}$-$\PsAlg$. 

\textbf{Future task (assume this works):} Prove the equivalence of operads/algebras/pseudoalgebras with the original permuativity $G$-operad.

\hfil

\textbf{Current Goal:} Add in extra axioms, so that there exist two $2$-functors 
\begin{align*}
    \RR_G:\Q_G\text{-}\PsAlg \leftrightarrows \F_{G,\RR,\_}\text{-}\PsAlg:\QQ_G
\end{align*}
inducing an isomorphism of (strict) 2-categories. Here, $\_$ is some (yet to be determined) axiom.

\section{Preliminaries (ND)}
\subsection{Symmetric Monoidal Categories}

\begin{defn}
    A category $\mathcal{V}$ is a \emph{monoidal category}, if there is a bifunctor $\tensor:\mathcal{V}\times\mathcal{V}\rightarrow\mathcal{V}$ (called \emph{monoidal product}), and an object $\kappa\in\mathcal{V}$ (called \emph{unit/monoidal unit}), together with the following natural isomorphisms:
    \begin{itemize}
        \item The \emph{associator} $a:(\_\tensor \_)\tensor \_ \xrightarrow{\sim} \_\tensor(\_\tensor \_)$ of functors $\mathcal{V}\times\mathcal{V}\times\mathcal{V}\xrightarrow{\sim}\mathcal{V}$. (So, $(X\tensor Y)\tensor Z\cong X\tensor(Y\tensor Z)$ in a canoncial way).
        \item The \emph{left unit} $\lambda:\kappa\tensor\_\xrightarrow{\sim} \Id_{\mathcal{V}}$ ($\kappa\tensor X\cong X$ in a canonical way).
        \item The \emph{right unit} $\rho:\_\tensor \kappa\xrightarrow{\sim}\Id_\mathcal{V}$ ($X\tensor \kappa\cong X$ in a canonical way).
    \end{itemize} 

    They also satisfy two extra axioms:
    \begin{itemize}
        \item \emph{Triangle Axiom:}
        \[\xymatrix{
            (X\tensor \kappa)\tensor Y \ar[dr]_-{\rho_X \tensor \id_Y} \ar[rr]^-{a_{X,\kappa,Y}} && X\tensor (\kappa\tensor Y) \ar[dl]^-{\id_X\tensor \lambda_Y}\\
            & X\tensor Y
        }\]
        \item \emph{Pentagon Axiom:}
        \[\xymatrix{
            ((X\tensor Y)\tensor Z)\tensor W \ar[rr]^-{a_{X\tensor Y,Z,W}} \ar[d]_-{a_{X,Y,Z}\tensor \id_W} && (X\tensor Y)\tensor(Z\tensor W) \ar[d]^-{a_{X,Y,Z\tensor W}}\\
            (X\tensor (Y\tensor Z))\tensor W \ar[dr]_-{a_{X,Y\tensor Z,W}} && X\tensor (Y\tensor (Z\tensor W)) \\
            & X\tensor ((Y\tensor Z)\tensor W) \ar[ur]_-{\id_X\tensor a_{Y,Z,W}}
        }\]
    \end{itemize}
\end{defn}
Based on \emph{Maclane's Coherence Theorem}, the above two axioms guarantees all finite sequences of $a,\lambda,\rho$ and their inverses agree, namely there is a unique isomorphism $P_1\rightarrow P_2$, where $P_i$ are (possibly) different orders of performing product on $X_1,...,X_n\in\mathcal{V}$. For reference, cf. \emph{Category Theory for Working Mathematicians}.

This guarantees all monoidal category to be equivalent to a \emph{strict monoidal category},namely all $a,\lambda,\rho$ above are ``identity transformations'', or $1\tensor X=X,X\tensor 1=X$, and $(X\tensor Y)\tensor Z=X\tensor (Y\tensor Z)$, so WLOG one can assume the product $\tensor$ is associative.

\begin{defn}
    A monoidal category $\mathcal{V}$ is \emph{braided}, if there is a natural isomorphism $B:\__{1} \tensor \__{2}\xrightarrow{\sim} \__{2}\tensor \__{1}$ (so $B_{X,Y}:X\tensor Y\xrightarrow{\sim}Y\tensor X$ is an isomorphism), together with the \emph{Hexagon axioms} satisfied:
    \[\xymatrix{
        (X\tensor Y)\tensor Z \ar[r]^-{a_{X,Y,Z}} \ar[d]_-{B_{X,Y}\tensor \Id_Z} & X\tensor (Y\tensor Z)\ar[r]^-{B_{X,Y\tensor Z}} & (Y\tensor Z)\tensor X \ar[d]^-{a_{Y,Z,X}}\\
        (Y\tensor X)\tensor Z \ar[r]_-{a_{Y,X,Z}} & Y\tensor (X\tensor Z) \ar[r]_-{\id_Y\tensor B_{X,Z}} & Y\tensor (Z\tensor X)
    }\]
    \[\xymatrix{
        X\tensor (Y\tensor Z) \ar[d]_-{\id_X\tensor B_{Y,Z}} & (X\tensor Y)\tensor Z \ar[l]_-{a_{X,Y,Z}} \ar[r]^-{B_{X\tensor Y,Z}} & Z\tensor (X\tensor Y)\\
        X\tensor (Z\tensor Y) & (X\tensor Z)\tensor Y \ar[l]^-{a_{X,Z,Y}} \ar[r]_-{B_{X,Z}\tensor \id_Y} & (Z\tensor X)\tensor Y \ar[u]_-{a_{Z,X,Y}}
    }\]
    $\mathcal{V}$ is a \emph{symmetric monoidal category}, if $B_{Y,X}=B_{X,Y}^{-1}$ for all $X,Y\in\mathcal{V}$.
\end{defn}
The two hexagons are uniqueness of 3-cycle permutations on three consecutive objects in a product, hence together with $B$ serving as transpositions, it guarantees unique isomorphism between all permutations of finite products, i.e., it imposes that there is a single way of doing such permutation (so there's no ``nontrivial braiding'', when $B_{Y,X}\circ B_{X,Y}:X\tensor Y\rightarrow Y\tensor X\rightarrow X\tensor Y$ is not identity). From my naive understanding of coherence theorem, WLOG we can again treat $X\tensor Y=Y\tensor X$.

\subsection{Operads, Operadic Algebras}

\begin{defn}
Given symmetric monoidal category $\mathcal{V}$, a \emph{$\Sigma$ operad} $\mathcal{C}$ in $\mathcal{V}$ is a collection of objects $\{\mathcal{C}(j)\in\mathcal{V}\}_{j\in\NN}$, each $\mathcal{C}(j)$ with a right $\Sigma_j$-action (a group homomorphism $\Sigma_j^\Op\rightarrow \Aut_\mathcal{V}(\mathcal{C}(j))$), a \emph{unit map} $\eta:1\rightarrow \mathcal{C}(1)$, together with composition maps
\[\gamma:\mathcal{C}(k)\tensor \mathcal{C}(j_1)\tensor...\tensor \mathcal{C}(j_k)\rightarrow \mathcal{C}(j_1+...+j_k)\]
that satisfies the following axioms:
\begin{itemize}
    \item \emph{Unit:}
    \[\xymatrix{
        \mathcal{C}(j)\tensor \kappa^{\tensor j} \ar[dr]_-{\id_{\mathcal{C}(j)}\tensor \eta^{\tensor j}} \ar[rr]^-{\sim} && \mathcal{C}(j)\\
        & \mathcal{C}(j)\tensor \mathcal{C}(1)^{\tensor j} \ar[ur]_-{\gamma}
    },\quad\quad \xymatrix{
        \kappa\tensor \mathcal{C}(j) \ar[dr]_-{\eta\tensor \id_{\mathcal{C}(j)}} \ar[rr]^-{\sim} && \mathcal{C}(j)\\
        & \mathcal{C}(1)\tensor \mathcal{C}(j) \ar[ur]_-{\gamma}
    }\]
    \item \emph{Associativity:} Fix natural numbers $k$, $j = \sum_{s=1}^k j_s$, define index $g_s:= j_1+...+j_s$ for $1\leq s\leq k$. For each $i_1,...,i_{g_k}=i_j$, with sum $i=\sum_{r=1}^j i_r$, and $h_s:= i_{g_{(s-1)}+1}+...+i_{g_s}$ (so $i=\sum_{s=1}^k h_s$ also), the following diagram commutes:
    \[\xymatrix{
        \left[\mathcal{C}(k)\tensor \left(\bigtensor_{s=1}^k \mathcal{C}(j_s)\right)\right]\tensor \left(\bigtensor_{r=1}^j \mathcal{C}(i_r)\right) \ar[dd]_-{\text{shuffle}} \ar[rr]^-{\gamma\tensor \id} && \mathcal{C}(j) \tensor \left(\bigtensor_{r=1}^j \mathcal{C}(i_r)\right) \ar[d]^-{\gamma}\\
        && \mathcal{C}(i)\\
        \mathcal{C}(k) \tensor \left[\bigtensor_{s=1}^k \left(\mathcal{C}(j_s)\tensor \left(\bigtensor_{q=1}^{j_s} \mathcal{C}(i_{g_{(s-1)}+q})\right)\right)\right] \ar[rr]_-{\id_{\mathcal{C}(k)} \tensor (\bigtensor_s \gamma)} && \mathcal{C}(k)\tensor \left(\bigtensor_{s=1}^k \mathcal{C}(h_s)\right) \ar[u]_-{\gamma}
    }\]
    \item \emph{Equivariance:}
    \begin{itemize}
        \item Given $\sigma\in\Sigma_k$, $j_1+...+j_k = \Sigma_j$, let $\sigma(j_1,...,j_k)\in\Sigma_j$ denotes ``permutation of $k$-blocks of letters'', by $(j_1,...,j_k)\mapsto (j_{\sigma^{-1}(1)},...,j_{\sigma^{-1}(k)})$.
        \begin{exmp}
            Given $j=5=2+3$, the permutation $(1\ 2)\in \Sigma_2$, $\sigma(2,3)$ is the permutation $\langle 1,2;3,4,5 \rangle \mapsto \langle 3,4,5;1,2\rangle$ (namely changing the blocks' order, without changing the order in each block).
        \end{exmp} 
        Then, the following diagram commutes:
        \[\xymatrix{
            \mathcal{C}(k)\tensor \left(\bigtensor_{r=1}^k \mathcal{C}(j_r)\right) \ar[d]_-{\gamma} \ar[rr]^-{\sigma\tensor \text{ shuffle}_{\sigma^{-1}}} && \mathcal{C}(k)\tensor \left(\bigtensor_{r=1}^k \mathcal{C}(j_{\sigma(r)})\right) \ar[d]^-{\gamma}\\
            \mathcal{C}(j) \ar[rr]_-{\sigma(j_{\sigma(1)},...,j_{\sigma(k)})} && \mathcal{C}(j)
        }\]
        \item Given $\tau_r\in \Sigma_{j_r}$, denote $\tau_1\oplus ... \oplus \tau_k\in \Sigma_j$ by letting each $\tau_r$ permuting the $j_r$th block in $j$ (for instance, let $\tau_1$ acts on $1,...,j_1$; let $\tau_2$ acts on $j_1+1,...,j_1+j_2$, etc.). Then, the following diagram commutes:
        \[\xymatrix{
            \mathcal{C}(k)\tensor \left(\bigtensor_{r=1}^k \mathcal{C}(j_r)\right)\ar[d]_-{\gamma} \ar[rr]^-{\id_{\mathcal{C}(k)}\tensor (\bigtensor_r \tau_r)} && \mathcal{C}(k)\tensor \left(\bigtensor_{r=1}^k \mathcal{C}(j_r)\right) \ar[d]^-{\gamma}\\
            \mathcal{C}(j) \ar[rr]_{\tau_1\oplus...\oplus\tau_k} && \mathcal{C}(j)
        }\]
    \end{itemize}
\end{itemize}

If $\mathcal{C}(0)=\kappa$ (with $\gamma:\mathcal{C}(0)\rightarrow \mathcal{C}$ being $\id_{\mathcal{C}(0)}$), the operad is \emph{reduced}.
\end{defn}

\begin{defn}
    Let $\mathcal{C},\mathcal{C}'$ be two operads on symmetric monoidal category $\mathcal{V}$, a \emph{morphism of operads} $\psi:\mathcal{C}\rightarrow \mathcal{C}'$, is a sequence of $\Sigma_j$-equivariant morphisms $\psi_j:\mathcal{C}(j)\rightarrow \mathcal{C}'(j)$ in $\mathcal{V}$ (i.e., $\sigma^{-1}\circ \psi_j\circ \sigma = \psi_j$, for all $\sigma\in \Sigma_j$), such that it preserves the operad's structure morphism:
    \[\xymatrix{
        \mathcal{C}(k)\tensor \mathcal{C}(j_1)\tensor...\tensor \mathcal{C}(j_k) \ar[d]_-{\psi_k\tensor \psi_{j_1}\tensor...\tensor \psi_{j_k}} \ar[r]^-{\gamma} & \mathcal{C}(j_1+...+j_k) \ar[d]^-{\psi_{j_1+...+j_k}}\\
        \mathcal{C}'(k)\tensor \mathcal{C}'(j_1)\tensor ..\tensor \mathcal{C}'(j_k) \ar[r]_-{\gamma'} & \mathcal{C}'(j_1+...+j_k)
    }\]
\end{defn}

\begin{defn}
    Given an operad $\mathcal{C}$ over a symmetric monoidal category $\mathcal{V}$, a \emph{$\mathcal{C}$-algebra} is an object $A\in\mathcal{V}$, with collections of morphisms $\theta:\mathcal{C}(j)\tensor A^j\rightarrow A$, that satisfies the following axioms:
    \begin{itemize}
        \item \emph{Associativity:} Given $j = \sum_{i=1}^k j_i$, the following diagram commutes:
        \[\xymatrix{
            \left[C(k)\tensor \left(\bigtensor_{i=1}^k C(j_i)\right)\right] \tensor A^{\tensor j} \ar[dd]_-{\text{shuffle}} \ar[rr]^-{\gamma\tensor \id_{A^{\tensor j}}} && C(j)\tensor A^{\tensor j} \ar[dr]^-{\theta}\\
            & & & A\\
            C(k)\tensor \left[\bigtensor_{i=1}^k \left(C(j_i)\tensor A^{\tensor j_i}\right)\right] \ar[rr]_-{\id_{C(k)}\tensor \theta^k} && C(k)\tensor A^{\tensor k} \ar[ur]_-{\theta}
        }\]
        (Note: This part uses the isomorphism $A^{\tensor j}\cong\bigtensor_{i=1}^k A^{\tensor j_i})$.
        \item \emph{Unit:} The following diagram commutes:
        \[\xymatrix{
            \kappa \tensor A \ar[dr]_-{\eta\tensor \id_A} \ar[rr]^-{\sim} && A\\
            & C(1)\tensor A \ar[ur]_-{\theta}
        }\]
        \item \emph{Eauivariance:} Given any permutation $\sigma\in\Sigma_j$, the following diagram commutes:
        \[\xymatrix{
            C(j)\tensor A^{\tensor j} \ar[dr]_-{\theta} \ar[rr]^-{\sigma\tensor \text{ shuffle}_{\sigma}} && C(j)\tensor A^{\tensor j} \ar[dl]^-{\theta}\\
            & A
        }\]
    \end{itemize}
\end{defn}

\subsection{$G$-Operad $\mathscr{Q}_G$ in $\Cat_G$}

\begin{defn}
    Let $Q_G$ be the suboperad of $P_G$, generated by all the group homomorphisms $\alpha:G\rightarrow\Sigma_n$ spanning through all $n\in\NN$, the $P_G$ structure maps $\gamma_G$, and the $(G\times\Sigma)$-action on the elements. Define the operad $\Q_G:= \mathscr{E}Q_G$ as the chaotic generalization in $\Cat_G$.
\end{defn}

This operad is designed as a smallest suboperad that contains all group homomorphisms from $G$ to a symmetric group, such that it's still a free $(G\times\Sigma)$-operad.

\textbf{Potential (fun) question:} Is this suboperad finitely generated, or equals to any $\Q_{G,\mathfrak{D}}$ described in \cite{BBKO21}?

\begin{warn}
    If naively taken all group homomorphisms and their orbits under $(G\times\Sigma)$-action, it doesn't form a suboperad in general for nontrivial group $G$. A counterexample can be constructed even for $G=\ZZ/2\ZZ$.
\end{warn}

\subsection{The 2-Category $\Q_G$-$\PsAlg$}

We mostly follow \cite{LMRZ26} in defining $\Q_G$-pseudoalgebras.

\begin{defn}[$\Q_G$-pseudoalgebra]
    A $\Q_G$-pseudoalgebra is a $G$-category $\A$ and operad action functors
    \begin{align*}
        \theta_j:\Q_G(j)\times \A^j\rightarrow\A
    \end{align*}
    for each $n\in\NN$, together with an invertible natural transformation $\phi_1$ in the diagram
    \begin{align}
        \xymatrix{ 
        \A\ar[r]^{\iota}\ar[dr]_{\Id_\A} & \Q_G(1)\times\A\ar[d]^{\theta_1}\ar@{=>}[dl]+<26pt>^>>>>{\phi_1}\\ 
        & C
        }
    \end{align}
    and a natural transformation $\phi=\phi(k;j_1,...,j_k)$ for each diagram 
    \begin{align}
        \xymatrix{
            \Q_G(k) \times \left(\prod_{r=1}^k \Q_G(j_r)\times \A^{j_r}\right) \ar[dd]_-{\text{shuffle}} \ar[rr]^-{\id\times\prod_r\theta_{j_r}} && \Q_G(k)\times\A^k \ar[d]^-{\theta_k} \ar@{=>}[ddll]^-{\phi}\\
            && \A\\
            \Q_G(k)\times \prod_{r=1}^k\Q_G(j_r)\times \A^{j_+} \ar[rr]_-{\gamma\times\id} && \Q_G(j_+)\times\A^{j_+} \ar[u]_-{\theta_{j_+}}
        }
    \end{align}
    such that the following strict equivariance hold for $\theta_j$ and $\phi$:
    \begin{itemize}
        \item[(i)]\textbf{(Equivariance of $\theta_j$)} The following diagram commutes for $(g,\sigma)\in G\times\Sigma_j$:
        \begin{align}
            \xymatrix{
                \Q_G(j)\times \A^j \ar[dr]_-{\theta_j}  \ar[rr]^-{(g,\sigma)\times (g,\sigma)^{-1}} && \Q_G(j)\times \A^j \ar[dl]^-{\theta_j}\\
                & \A
            }
        \end{align}
        \item[(ii)]\textbf{(Equivariance of $\phi$)} The following equalities hold when given $g\in G$, $\sigma\in\Sigma_k$, and $\tau_r\in\Sigma_{j_r}$:
        \begin{align*}
            \phi(k;j_1,...,j_k) = \phi(k;j_{\sigma(1)},...,j_{\sigma_k}) \circ ((g,\sigma)\times (g,\sigma)^{-1})
        \end{align*}
        \begin{align*}
            \phi(k;j_1,...,j_k)=\phi(k;j_1,...,j_k)\circ (\id\times \prod_r(\tau_r\times \tau_r^{-1}))
        \end{align*}
    \end{itemize}
    They also need to satisfy the following coherence properties:
    \begin{itemize}
        \item[(iii)] \textbf{(Operadic Composition)} The following pasting diagrams are equal, given $j_+=\sum j_r$, $i_r = \sum_si_{r,s}$, and $i_+ = \sum_{r,s}i_{r,s}$ with $1\leq r\leq k$ and $1\leq s\leq j_r$:
        \item[] \begin{align*}
            \xymatrix{
                \Q_G(k)\times \prod_r\left(\Q_G(j_r)\times \prod_s\left(\Q_G(i_{r,s})\times \A^{i_{r,s}}\right)\right) \ar[r]^-{\id\times \prod(\id\times \theta_{i_{r,s}})} \ar[dd]_{\gamma\times\id} & Q_G(k)\times \prod_r\left(\Q_G(j_r)\times \A^{j_r}\right) \ar[dd]_-{\gamma} \ar[dr]^-{\id\times \prod_{\theta_{j_r}}}\\
            && \Q_G(k)\times \A^k \ar[dd]^-{\theta_k} \ar@{=>}[dl]^-{\phi}\\
            \Q_G(j_+)\times \prod_{r,s}\left(\Q_G(i_{r,s})\times \A^{i_{r,s}}\right) \ar[r]^-{\id\times \prod \theta_{i_{r,s}}} \ar[dr]_-{\gamma\times \id} & \Q_G(j_+)\times \A^{j_+} \ar[dr]^-{\theta_{j_+}} \ar@{=>}[d]^-{\phi}\\
            & \Q_G(i_+)\times \A^{i_+} \ar[r]_-{\theta_{j_+}} & \A
            }
        \end{align*}
        equals
        \begin{align*}
            \xymatrix{
                \Q_G(k)\times \prod_r\left(\Q_G(j_r)\times \prod_s\left(\Q_G(i_{r,s})\times\A^{i_{r,s}}\right)\right) \ar[r]^-{\id\times \prod(\id\times \prod \theta_{i_{r,s}})} \ar[dd]_-{\gamma\times\id} \ar[dr]_-{\id\times \prod(\gamma\times\id)} & \Q_G(k)\times \prod_r\left(\Q_g(j_r)\times \A^{j_r}\right) \ar[dr]^-{\id\times \prod\theta_{j_r}} \ar@{=>}[d]^-{\id\times \prod\phi}\\
                & \Q_G(k)\times \prod_r\left(\Q_G(i_r)\times \A^{i_r}\right) \ar[r]_-{\id\times\prod\gamma} \ar[dd]_-{\gamma\times\id} & \Q_G(k)\times\A^k \ar[dd]^-{\theta_k} \ar@{=>}[ddl]^-{\phi}\\
                \Q_G(j_+)\times\prod_{r,s}\left(\Q_G(i_{r,s})\times \A^{i_{r,s}}\right) \ar[dr]_-{\gamma\times\id}\\
                & \Q_G(i_+)\times\A^{i_+} \ar[r]_-{\theta_{i_+}} & \A
            }
        \end{align*}
        (commuting squares are operadic associativity)
        \item[(iv)]\textbf{(Operadic Identity)} The following two pairs of two pasting diagrams are equal:
        \begin{align*}
            \xymatrix{
                \Q_G(k)\times \A^k\ar[r]^-{\id\times\iota^k} & \Q_G(k)\times \left(\Q_G(1)\times\A\right)^k \ar[r]^-{\id\times \theta_1^k} \ar[d]_-{\text{shuffle}} & \Q_G(k)\times \A^k \ar[dd]^-{\theta_k} \ar@{=>}[ddl]^-{\phi}\\
                & \Q_G(k)\times \Q_G(1)^k\times \A^k \ar[d]_-{\gamma\times\id}\\
                &\Q_G(k)\times\A^k\ar[r]_-{\theta_k} & \A
            }
        \end{align*}
        equals
        \begin{align*}
            \xymatrix{
                & \Q_G(k)\times \left(\Q_G(1)\times\A\right)^k \ar[dr]^-{\id\times\theta_1^k} \ar@{=>}[d]^-{\id\times \prod \phi}\\
                \Q_G(k)\times\A^k\ar[ur]^-{\id\times\iota^k} \ar[rr]_-{\id} \ar[d]_-{\id\times\iota^k} \ar[dr]_-{\iota^k} && \Q_G(k)\times\A^k\ar[r]^-{\theta_k} & \A\\
                \Q_G(k)\times\left(\Q_G(1)\times\A\right)^k \ar[r]_-{\text{shuffle}}& \Q_G(k)\times \Q_G(1)^k\times\A^k \ar[ur]_-{\gamma\times\id}
            }
        \end{align*}
        (uses operadic associativity)

        Also:
        \begin{align*}
            \xymatrix{
                \Q_G(k)\times\A^k\ar[r]^-{\iota} \ar[dr]_-{\id} & \Q_G(1)\times\Q_G(k)\times \A^k \ar[r]^-{\id\times\theta_k} \ar[d]_-{\gamma\times\id} & \Q_G(1)\times\A \ar[d]^-{\theta_1} \ar@{=>}[dl]^-{\phi}\\
                &\Q_G(k)\times\A^k \ar[r]_-{\theta_k} & \A
            }
        \end{align*}
        equals
        \begin{align*}
            \xymatrix{
                \Q_G(k)\times \A^k \ar[r]^-{\iota} \ar[d]_-{\theta_k} & \Q_G(1)\times\Q_G(k)\times \A^k \ar[d]^-{\id\times\theta_k}\\
                \A \ar[r]^-{\iota} \ar[dr]_-{\id} & \Q_G(1)\times\A \ar[d]^-{\theta_1} \ar@{=>}[dl]+<30pt>^>>>>>>>{\phi_1}\\
                & \A
            }
        \end{align*}
        (Uses operadic unitality)
    \end{itemize}
\end{defn}

\begin{defn}[$\Q_G$-pseudomorphism]
    A $\Q_G$-pseudomorphism $(\FF,\zeta):(\A,\theta,\phi)\rightarrow(\B,\vartheta,\varphi)$ consists of \emph{$G$-functor} $\FF:\A\rightarrow\B$, such that for each $j$ in the follogin diagram commutes up to an invertibal natural transformation $\zeta_j$:
    \begin{align*}
        \xymatrix{
            \Q_G(j)\times \A^j\ar[r]^-{\id\times\FF^j} \ar[d]_-{\theta_j} & \Q_G(j)\times \B^j \ar[d]^-{\vartheta_j} \ar@{=>}[dl]^-{\zeta_j}\\
            \A \ar[r]_-{\FF} & \B
        }
    \end{align*}
    Moreover, they satisfy the following properties:
    \begin{itemize}
        \item[(i)]\textbf{(Equivariance of $\zeta$)} For each $j$,$\zeta_j=\zeta_j\circ ((g,\sigma)\times (g,\sigma)^{-1})$ for all $(g,\sigma)\in G\times\Sigma_j$.
        \item[(ii)] \textbf{(Preserves Identity)} The following two pasting diagrams are equal:
        \begin{align*}
            \xymatrix{
                \A \ar[r]^-{\FF} \ar[d]_-{\iota} & \B \ar[d]_-{\iota} \ar@(r,u)[ddr]^-{\id}\\
                \Q_G(1)\times\A \ar[r]^-{\id\times\FF} \ar[dr]_-{\theta_1} & \Q_G(1)\times\B \ar[dr]_-{\vartheta_1} \ar@{=>}[r]-<10pt>^-{\varphi_1} & \\
                & \A \ar[r]_-{\FF} \ar@{=>}[u]_-{\zeta_1^{-1}} & \B
            }\quad = \quad \xymatrix{
                \A \ar[r]^-{\FF} \ar[d]_-{\iota} \ar[dr]^-{\id} & \B \ar[dr]^-{\id}\\
                \Q_G(1)\times \A \ar@{=>}[ur]-<25pt>^-{\phi_1} \ar[r]_-{\theta_1} & \A \ar[r]_-{\FF} & \B
            }
        \end{align*}
        \item[(iii)] \textbf{(Preserves Composition)} The following two pasting diagrams are equal:
        \begin{align*}
            \xymatrix{
                \Q_G(k)\times \prod_r\left(\Q_G(j_r)\times \A^{j_r}\right) \ar[rr]^-{\id\times\prod(\id\times \FF^{j_r})} \ar[dr]_-{\id\times \prod\theta_{j_r}} \ar[dd]_-{\gamma} && \Q_G(k)\times \prod_r\left(\Q_G(j_r)\times \B^{j_r}\right) \ar[dr]^-{\id\times \prod\vartheta_{j_r}} \ar@{=>}[dl]^-{\id\times \prod \zeta_{j_r}}\\
                & \Q_G(k)\times \A^k \ar[rr]_-{\id\times \FF^k} \ar@{=>}[dl]^-{\phi} \ar[dd]_-{\theta_k} && \Q_G(k)\times \B^k \ar[dd]^-{\vartheta_k} \ar@{=>}[ddll]^-{\zeta_k}\\
                \Q_G(j)\times \A^j \ar[dr]_-{\theta_j}\\
                & \A \ar[rr]_-{\FF} && \B
            }
        \end{align*}
        equals
        \begin{align*}
            \xymatrix{
                \Q_G(k)\times \prod_r\left(\Q_G(j_r)\times\A^{j_r}\right) \ar[rr]^-{\id\times \prod(\id\times \FF^{j_r})} \ar[dd]_-{\gamma} && \Q_G(k)\times \prod\left(\Q_G(j_r)\times \B^{j_r}\right) \ar[dd]_-{\gamma} \ar[dr]^-{\id\times \prod\theta_{j_r}}\\
                &&& \Q_G(k)\times \B^k \ar[dd]^-{\vartheta_k} \ar@{=>}[dl]^-{\varphi}\\
                \Q_G(j)\times\A^j \ar[rr]^-{\id\times \FF^j} \ar[dr]_-{\theta_j} && \Q_G(j)\times \B^j \ar[dr]^-{\vartheta_j} \ar@{=>}[dl]^-{\zeta_j}\\
                & \A \ar[rr]_-{\FF} && \B
            }
        \end{align*}
    \end{itemize}
\end{defn}

\section{Counterexample: $\F_{G,\RR}$-pseudoalgebra that's not $\Q_G$-pseudoalgebra}

Here we fix the group $G:=\ZZ/2\ZZ$. Which, we have $\Q_G(1)=\cP_G(1) = *$, in particular $G\times \Sigma_1\cong G$ has trivial action on these. This leads to the following properties of $\Q_G$ or $\cP_G$-pseudoalgebra:

\begin{prop}
    For any $\Q_G$-pseudoalgebra $\A$, any $G$-action $g:\A\xrightarrow{\sim} \A$ is naturally isomorphic to the identity functor $\Id_\A$.
\end{prop}
\begin{proof}
    Pasting the unitality and equivariance diagram together, we get the following naturality diagram:
    \begin{align}
        \xymatrix{
            \A \ar@(dl,l)[ddr]_-{\Id_\A} \ar[dr]^-{\iota} \ar[rrrr]^-{g^{-1}} &&&& \A \ar[dl]_-{\iota} \ar@(dr,r)[ddl]^-{\Id_\A}\\
            &\Q_G(1)\times\A \ar@{=>}[l]^-{\phi_1} \ar[d]_-{\theta_1} \ar[rr]^-{g\times g^{-1}} && \Q_G(1)\times\A \ar[d]^-{\theta_1} \ar@{<=}[r]_-{\phi_1^{-1}} & \\
            &\A \ar@{=}[rr] && \A
        }
    \end{align}
    Here, the top trapezoid commutes because $G$ acts trivially on $\Q_G(1)$. This identifies a natural isomorphism as follow:
    \begin{align}
        g^{-1}=\Id_\A\circ g^{-1} \xRightarrow{\phi_1^{-1}*\id_{g^{-1}}} (\theta_1\circ\iota)\circ g^{-1} = \theta_1\circ (g\times g^{-1})\circ \iota = \theta_1\circ \iota \xRightarrow{\phi_1} \Id_\A
    \end{align}
\end{proof}
%Self note: In particular, this extends the $G$-action $G\rightarrow \Aut_{\Cat_G}(\A)$ into a functor $\mathscr{E}G\rightarrow \Aut_{\Cat_G}(\A)$.

\hfil

To construct a counterexample for our fixed $G$, it suffices to find an $\F_{G,\RR}$-pseudoalgebra that doesn't have the above property. Here, we largely use the fact that for two naturally isomorphic functors, if the target category is discrete, the two functors must be equal.

\begin{con}
Let $\A:=\{0,1\}$ denotes the discrete category with $2$ objects, which a $G$-action on $\A$ is specified by its action on the objects due to its discreteness. Define a $G$-action on $\A$ given by 
\begin{align}
    &H_0:\A\xrightarrow{\sim}\A,\quad H_0(0)=0,\ H_0(1)=1\\
    &H_1:\A\xrightarrow{\sim}\A,\quad H_1(0)=1,\ H_1(1)=0,
\end{align}
Which, $H_0^2=H_0$, $H_0\circ H_1=H_1\circ H_0$, and $H_1^2=H_0$, so $H_0,H_1$ satisfies the relations of $G=\ZZ/2\ZZ$, hence inducing a $G$-action on $\A$.

\hfil

Define $\tilde{\A}:\F_G\rightarrow \Cat_G$ given by $\tilde{\A}(\mathbf{n}^\alpha):=\A^{\mathbf{n}^\alpha}$ (with underlying category $\A^n$, a discrete category with $2^n$ objects), and for any morphism $f:\mathbf{n}^\alpha\rightarrow\mathbf{m}^\beta$, define $\tilde{\A}f:\A^{\mathbf{n}^\alpha}\rightarrow\A^{\mathbf{m}^\beta}$ by 
\begin{align}
    \tilde{\A}f\left((a_i)_{i=1,...,n}\right):= \left(\sum_{i\in f^{-1}(j)}a_i\ \mathrm{mod}\ 2 \right)_{j=1,...,m},
\end{align}
for special case where $f^{-1}(j)=\emptyset$, define the $j$th entry as $0$.

To check it preserves composition up to coherent natural isomorphism, consider $\mathbf{n}^\alpha\xrightarrow{f}\mathbf{m}^\beta\xrightarrow{g}\mathbf{k}^\gamma$ in $\F_G$, then $\tilde{\A}(g\circ f):\tilde{\A}^{\mathbf{n}^\alpha}\rightarrow\tilde{\A}^{\mathbf{m}^\beta}$ is given as follow:
\begin{align}
    \tilde{\A}(g\circ f)((a_i)_{i=1,...,n}) &= \left(\sum_{i\in (g\circ f)^{-1}(\ell)}a_i\ \mathrm{mod}\ 2\right)_{\ell=1,...,k}\\
    &= \left(\sum_{i\in f^{-1}(g^{-1}(\ell))}a_i\ \mathrm{mod}\ 2\right)_{\ell=1,...,k}\\
    &= \left(\sum_{j\in g^{-1}(\ell)}\left(\sum_{i\in f^{-1}(j)}a_i\ \mathrm{mod}\ 2\right)\ \mathrm{mod}\ 2\right)_{\ell=1,...,k}\\
    &= \tilde{\A}g\left(\left(\sum_{i\in f^{-1}(j)}a_i\ \mathrm{mod}\ 2\right)_{j=1,...,m}\right)\\
    &= \tilde{\A}g\left(\tilde{\A}f((a_i)_{i=1,...,k})\right)
\end{align}
(Here, the well-defineness of the function heavily relies on the assumption that if $f^{-1}(j)=\emptyset$, the corresponding sum is $0$). In particular, $\tilde{\A}(g\circ f) = \tilde{\A}g\circ \tilde{\A}f$, so $\tilde{\A}$ is a functor (or an $\F_G$-algebra), and since each $\tilde{\A}(\mathbf{n}^\alpha):= \A^{\mathbf{n}^\alpha}$, it's in fact an $\F_{G,\RR}$-algebra.

Yet, since $\A$ is discrete, with $H_0,H_1$ being distinct functors on $\A$, we see that not all the $G$-actions on $\A$ are naturally isomorphic to $\Id_\A$, violating the property described in previous proposition. Hence, $\A$ cannot be a $\Q_G$-pseudoalgebra, or even a $\cP_G$-pseudoalgebra due to the same reasoning.
\end{con}

\hfil

This shows that the functor $\RR_G:\cP_G$-$\PsAlg\rightarrow \F_G$-$\PsAlg$ is not essentially surjective, hence cannot be an equivalence of categories. Same statement holds true for $\Q_G$-$\PsAlg$.
