\section{Introduction (ND)}

Fix $G$ as a finite group. Let $P_G$ be the \emph{associativity $G$-operad} in $\Sets_G$, defined by $P_G(n):= \Sets(G,\Sigma_n)$, with structure maps defined by post composition
\begin{align*}
    \gamma:\Sets(G,\Sigma_k)\times \prod_{r=1}^k\Sets(G,\Sigma_{j_r})\cong \Sets\left(G,\Sigma_k\times \prod_{r=1}^k \Sigma_{j_r}\right)\xrightarrow{\ \gamma\circ\_\ }\Sets(G,\Sigma_j),
\end{align*},
where $j=\sum_{r=1}^k j_r$, and $\gamma:\Sigma_k\times \prod_{r=1}^k \Sigma_{j_r}\rightarrow \Sigma_j$ is the structure map for associativity operad $\mathscr{M}$ in $\Sets$. This is naturally a free $(G\times\Sigma)$-operad, as the right $(G\times \Sigma_n)$-action at each $P_G(n)$ is free.

However, it is mentioned in \cite{BBKO21} that such $G$-operad is not finitely generated, as a rough explanation, for every $n\in\NN$, there exists $m>n$ and a function $f:G\rightarrow \Sigma_m$, such that it cannot be generated using $\{P_G(k)\}_{k\leq n}$ with the operadic structure map and the $(G\times \Sigma)$-actions.

\hfil

As a substitution, \cite{BBKO21} defines a finitely-generated suboperad $Q_{G,\mathfrak{D}}\hookrightarrow P_G$ (for some parameter $\mathfrak{D}$ that's associated with choice of transitive $G$-sets), such that after chaotic generalization the two are equivalent in $\Cat_G$, and having equivalent categories of operadic algebras.

We'll mimic the statement, define an intermediate operad $Q_{G,\mathfrak{D}}\hookrightarrow Q_G\hookrightarrow P_G$, promote it to $\Q_G$ in $\Cat_G$, and modify the proof in \cite{LMRZ26} to prove the equivalence of 2-categories between $\Q_G$-$\PsAlg$ and $\F_{G,\RR}$-$\PsAlg$. 

\textbf{Future work: Prove the equivalence of operads/algebras/pseudoalgebras with the original permuativity $G$-operad}.

\section{$G$-Operad $\mathscr{Q}_G$ in $\Cat_G$}

\begin{defn}
    Let $Q_G$ be the suboperad of $P_G$, generated by all the group homomorphisms $\alpha:G\rightarrow\Sigma_n$ spanning through all $n\in\NN$, the $P_G$ structure maps $\gamma_G$, and the $(G\times\Sigma)$-action on the elements. Define the operad $\Q_G:= \mathscr{E}Q_G$ as the chaotic generalization in $\Cat_G$.
\end{defn}

This operad is designed as a smallest suboperad that contains all group homomorphisms from $G$ to a symmetric group, such that it's still a free $(G\times\Sigma)$-operad.

\begin{warn}
    If naively taken all group homomorphisms and their orbits under $(G\times\Sigma)$-action, it doesn't form a suboperad in general for nontrivial group $G$. A counterexample can be constructed even for $G=\ZZ/2\ZZ$.
\end{warn}

\subsection{The 2-Category $\Q_G$-$\PsAlg$}

We mostly follow \cite{LMRZ26} in defining $\Q_G$-pseudoalgebras.

\begin{defn}[$\Q_G$-pseudoalgebra]
    A $\Q_G$-pseudoalgebra is a $G$-category $\A$ and operad action functors
    \begin{align*}
        \theta_j:\Q_G(j)\times \A^j\rightarrow\A
    \end{align*}
    for each $n\in\NN$, together with an invertible natural transformation $\phi_1$ in the diagram
    \begin{align}
        \xymatrix{ 
        \A\ar[r]^{\iota}\ar[dr]_{\Id_\A} & \Q_G(1)\times\A\ar[d]^{\theta_1}\ar@{=>}[dl]+<26pt>^>>>>{\phi_1}\\ 
        & C
        }
    \end{align}
    and a natural transformation $\phi=\phi(k;j_1,...,j_k)$ for each diagram 
    \begin{align}
        \xymatrix{
            \Q_G(k) \times \left(\prod_{r=1}^k \Q_G(j_r)\times \A^{j_r}\right) \ar[dd]_-{\text{shuffle}} \ar[rr]^-{\id\times\prod_r\theta_{j_r}} && \Q_G(k)\times\A^k \ar[d]^-{\theta_k} \ar@{=>}[ddll]^-{\phi}\\
            && \A\\
            \Q_G(k)\times \prod_{r=1}^k\Q_G(j_r)\times \A^{j_+} \ar[rr]_-{\gamma\times\id} && \Q_G(j_+)\times\A^{j_+} \ar[u]_-{\theta_{j_+}}
        }
    \end{align}
    such that the following strict equivariance hold for $\theta_j$ and $\phi$:
    \begin{itemize}
        \item[(i)]\textbf{(Equivariance of $\theta_j$)} The following diagram commutes for $(g,\sigma)\in G\times\Sigma_j$:
        \begin{align}
            \xymatrix{
                \Q_G(j)\times \A^j \ar[dr]_-{\theta_j}  \ar[rr]^-{(g,\sigma)\times (g,\sigma)^{-1}} && \Q_G(j)\times \A^j \ar[dl]^-{\theta_j}\\
                & \A
            }
        \end{align}
        \item[(ii)]\textbf{(Equivariance of $\phi$)} The following equalities hold when given $g\in G$, $\sigma\in\Sigma_k$, and $\tau_r\in\Sigma_{j_r}$:
        \begin{align*}
            \phi(k;j_1,...,j_k) = \phi(k;j_{\sigma(1)},...,j_{\sigma_k}) \circ ((g,\sigma)\times (g,\sigma)^{-1})
        \end{align*}
        \begin{align*}
            \phi(k;j_1,...,j_k)=\phi(k;j_1,...,j_k)\circ (\id\times \prod_r(\tau_r\times \tau_r^{-1}))
        \end{align*}
    \end{itemize}
    They also need to satisfy the following coherence properties:
    \begin{itemize}
        \item[(iii)] \textbf{(Operadic Composition)} The following pasting diagrams are equal, given $j_+=\sum j_r$, $i_r = \sum_si_{r,s}$, and $i_+ = \sum_{r,s}i_{r,s}$ with $1\leq r\leq k$ and $1\leq s\leq j_r$:
        \item[] \begin{align*}
            \xymatrix{
                \Q_G(k)\times \prod_r\left(\Q_G(j_r)\times \prod_s\left(\Q_G(i_{r,s})\times \A^{i_{r,s}}\right)\right) \ar[r]^-{\id\times \prod(\id\times \theta_{i_{r,s}})} \ar[dd]_{\gamma\times\id} & Q_G(k)\times \prod_r\left(\Q_G(j_r)\times \A^{j_r}\right) \ar[dd]_-{\gamma} \ar[dr]^-{\id\times \prod_{\theta_{j_r}}}\\
            && \Q_G(k)\times \A^k \ar[dd]^-{\theta_k} \ar@{=>}[dl]^-{\phi}\\
            \Q_G(j_+)\times \prod_{r,s}\left(\Q_G(i_{r,s})\times \A^{i_{r,s}}\right) \ar[r]^-{\id\times \prod \theta_{i_{r,s}}} \ar[dr]_-{\gamma\times \id} & \Q_G(j_+)\times \A^{j_+} \ar[dr]^-{\theta_{j_+}} \ar@{=>}[d]^-{\phi}\\
            & \Q_G(i_+)\times \A^{i_+} \ar[r]_-{\theta_{j_+}} & \A
            }
        \end{align*}
        equals
        \begin{align*}
            \xymatrix{
                \Q_G(k)\times \prod_r\left(\Q_G(j_r)\times \prod_s\left(\Q_G(i_{r,s})\times\A^{i_{r,s}}\right)\right) \ar[r]^-{\id\times \prod(\id\times \prod \theta_{i_{r,s}})} \ar[dd]_-{\gamma\times\id} \ar[dr]_-{\id\times \prod(\gamma\times\id)} & \Q_G(k)\times \prod_r\left(\Q_g(j_r)\times \A^{j_r}\right) \ar[dr]^-{\id\times \prod\theta_{j_r}} \ar@{=>}[d]^-{\id\times \prod\phi}\\
                & \Q_G(k)\times \prod_r\left(\Q_G(i_r)\times \A^{i_r}\right) \ar[r]_-{\id\times\prod\gamma} \ar[dd]_-{\gamma\times\id} & \Q_G(k)\times\A^k \ar[dd]^-{\theta_k} \ar@{=>}[ddl]^-{\phi}\\
                \Q_G(j_+)\times\prod_{r,s}\left(\Q_G(i_{r,s})\times \A^{i_{r,s}}\right) \ar[dr]_-{\gamma\times\id}\\
                & \Q_G(i_+)\times\A^{i_+} \ar[r]_-{\theta_{i_+}} & \A
            }
        \end{align*}
        (commuting squares are operadic associativity)
        \item[(iv)]\textbf{(Operadic Identity)} The following two pairs of two pasting diagrams are equal:
        \begin{align*}
            \xymatrix{
                \Q_G(k)\times \A^k\ar[r]^-{\id\times\iota^k} & \Q_G(k)\times \left(\Q_G(1)\times\A\right)^k \ar[r]^-{\id\times \theta_1^k} \ar[d]_-{\text{shuffle}} & \Q_G(k)\times \A^k \ar[dd]^-{\theta_k} \ar@{=>}[ddl]^-{\phi}\\
                & \Q_G(k)\times \Q_G(1)^k\times \A^k \ar[d]_-{\gamma\times\id}\\
                &\Q_G(k)\times\A^k\ar[r]_-{\theta_k} & \A
            }
        \end{align*}
        equals
        \begin{align*}
            \xymatrix{
                & \Q_G(k)\times \left(\Q_G(1)\times\A\right)^k \ar[dr]^-{\id\times\theta_1^k} \ar@{=>}[d]^-{\id\times \prod \phi}\\
                \Q_G(k)\times\A^k\ar[ur]^-{\id\times\iota^k} \ar[rr]_-{\id} \ar[d]_-{\id\times\iota^k} \ar[dr]_-{\iota^k} && \Q_G(k)\times\A^k\ar[r]^-{\theta_k} & \A\\
                \Q_G(k)\times\left(\Q_G(1)\times\A\right)^k \ar[r]_-{\text{shuffle}}& \Q_G(k)\times \Q_G(1)^k\times\A^k \ar[ur]_-{\gamma\times\id}
            }
        \end{align*}
        (uses operadic associativity)

        Also:
        \begin{align*}
            \xymatrix{
                \Q_G(k)\times\A^k\ar[r]^-{\iota} \ar[dr]_-{\id} & \Q_G(1)\times\Q_G(k)\times \A^k \ar[r]^-{\id\times\theta_k} \ar[d]_-{\gamma\times\id} & \Q_G(1)\times\A \ar[d]^-{\theta_1} \ar@{=>}[dl]^-{\phi}\\
                &\Q_G(k)\times\A^k \ar[r]_-{\theta_k} & \A
            }
        \end{align*}
        equals
        \begin{align*}
            \xymatrix{
                \Q_G(k)\times \A^k \ar[r]^-{\iota} \ar[d]_-{\theta_k} & \Q_G(1)\times\Q_G(k)\times \A^k \ar[d]^-{\id\times\theta_k}\\
                \A \ar[r]^-{\iota} \ar[dr]_-{\id} & \Q_G(1)\times\A \ar[d]^-{\theta_1} \ar@{=>}[dl]+<30pt>^>>>>>>>{\phi_1}\\
                & \A
            }
        \end{align*}
        (Uses operadic unitality)
    \end{itemize}
\end{defn}

\hfil

\begin{defn}[$\Q_G$-pseudomorphism]
    A $\Q_G$-pseudomorphism $(\FF,\zeta):(\A,\theta,\phi)\rightarrow(\B,\vartheta,\varphi)$ consists of \emph{$G$-functor} $\FF:\A\rightarrow\B$, such that for each $j$ in the follogin diagram commutes up to an invertibal natural transformation $\zeta_j$:
    \begin{align*}
        \xymatrix{
            \Q_G(j)\times \A^j\ar[r]^-{\id\times\FF^j} \ar[d]_-{\theta_j} & \Q_G(j)\times \B^j \ar[d]^-{\vartheta_j} \ar@{=>}[dl]^-{\zeta_j}\\
            \A \ar[r]_-{\FF} & \B
        }
    \end{align*}
    Moreover, they satisfy the following properties:
    \begin{itemize}
        \item[(i)]\textbf{(Equivariance of $\zeta$)} For each $j$,$\zeta_j=\zeta_j\circ ((g,\sigma)\times (g,\sigma)^{-1})$ for all $(g,\sigma)\in G\times\Sigma_j$.
        \item[(ii)] \textbf{(Preserves Identity)} The following two pasting diagrams are equal:
        \begin{align*}
            \xymatrix{
                \A \ar[r]^-{\FF} \ar[d]_-{\iota} & \B \ar[d]_-{\iota} \ar@(r,u)[ddr]^-{\id}\\
                \Q_G(1)\times\A \ar[r]^-{\id\times\FF} \ar[dr]_-{\theta_1} & \Q_G(1)\times\B \ar[dr]_-{\vartheta_1} \ar@{=>}[r]-<10pt>^-{\varphi_1} & \\
                & \A \ar[r]_-{\FF} \ar@{=>}[u]_-{\zeta_1^{-1}} & \B
            }\quad = \quad \xymatrix{
                \A \ar[r]^-{\FF} \ar[d]_-{\iota} \ar[dr]^-{\id} & \B \ar[dr]^-{\id}\\
                \Q_G(1)\times \A \ar@{=>}[ur]-<25pt>^-{\phi_1} \ar[r]_-{\theta_1} & \A \ar[r]_-{\FF} & \B
            }
        \end{align*}
        \item[(iii)] \textbf{(Preserves Composition)} The following two pasting diagrams are equal:
        \begin{align*}
            \xymatrix{
                \Q_G(k)\times \prod_r\left(\Q_G(j_r)\times \A^{j_r}\right) \ar[rr]^-{\id\times\prod(\id\times \FF^{j_r})} \ar[dr]_-{\id\times \prod\theta_{j_r}} \ar[dd]_-{\gamma} && \Q_G(k)\times \prod_r\left(\Q_G(j_r)\times \B^{j_r}\right) \ar[dr]^-{\id\times \prod\vartheta_{j_r}} \ar@{=>}[dl]^-{\id\times \prod \zeta_{j_r}}\\
                & \Q_G(k)\times \A^k \ar[rr]_-{\id\times \FF^k} \ar@{=>}[dl]^-{\phi} \ar[dd]_-{\theta_k} && \Q_G(k)\times \B^k \ar[dd]^-{\vartheta_k} \ar@{=>}[ddll]^-{\zeta_k}\\
                \Q_G(j)\times \A^j \ar[dr]_-{\theta_j}\\
                & \A \ar[rr]_-{\FF} && \B
            }
        \end{align*}
        equals
        \begin{align*}
            \xymatrix{
                \Q_G(k)\times \prod_r\left(\Q_G(j_r)\times\A^{j_r}\right) \ar[rr]^-{\id\times \prod(\id\times \FF^{j_r})} \ar[dd]_-{\gamma} && \Q_G(k)\times \prod\left(\Q_G(j_r)\times \B^{j_r}\right) \ar[dd]_-{\gamma} \ar[dr]^-{\id\times \prod\theta_{j_r}}\\
                &&& \Q_G(k)\times \B^k \ar[dd]^-{\vartheta_k} \ar@{=>}[dl]^-{\varphi}\\
                \Q_G(j)\times\A^j \ar[rr]^-{\id\times \FF^j} \ar[dr]_-{\theta_j} && \Q_G(j)\times \B^j \ar[dr]^-{\vartheta_j} \ar@{=>}[dl]^-{\zeta_j}\\
                & \A \ar[rr]_-{\FF} && \B
            }
        \end{align*}
    \end{itemize}
\end{defn}

\hfil

\begin{defn}[$\Q_G$-2-cells]
    Let $(\FF,\zeta),(\GG,\xi):(\A,\theta,\phi)\rightarrow (\B,\vartheta,\varphi)$ be two $\Q_G$-pseudomorphisms, a $\Q_G$-2-cell $\lambda:(\FF,\zeta)\Rightarrow(\GG,\xi)$ is a natural transformation $\lambda:\FF\rightarrow\GG$, such that the following two pasting diagrams are equal:
    \begin{align*}
        \xymatrix{
            \Q_G(j)\times \A^j \ar@(ur,ul)[r]^-{\id\times \FF^j} \ar@{=>}[r]^-{\id\times \lambda^j} \ar@(dr,dl)[r]^-{\id\times\GG^j} \ar[d]_-{\theta_j} & \Q_G(j)\times \B^j \ar[d]^-{\vartheta_j} \ar@{=>}[dl]^-{\xi_j}\\
            \A \ar[r]_-{\GG} & \B
        }\quad = \quad \xymatrix{
            \Q_G(j)\times \A^j \ar[d]_-{\theta_j} \ar[r]^-{\id\times \FF^j} & \Q_G(j)\times\B^j \ar[d]^-{\vartheta_j} \ar@{=>}[dl]_-{\zeta_j}\\
            \A \ar@(ur,ul)[r]_-{\FF} \ar@{=>}[r]_-{\lambda} \ar@(dr,dl)[r]_-{\GG} & \B
        }
    \end{align*}
\end{defn}

\section{2-Category $\F_G$-$\PsAlg$}

In the category $\F_G$ (all based finite $G$-sets), the homsets $\F_G(\mathbf{n}^\alpha,\mathbf{m}^\beta)$ can also be viewed as a discrete category. We'll use this identification to define some notions.

\begin{defn}[$\F_G$-pseudoalgebra]
    An $\F_G$-pseudoalgebra, is a pseudofunctor $\tilde{\A}:\F_G\rightarrow \Cat_G$, such that restricting to the full subcategory $\Pi_G\subset\F_G$, $\tilde{\A}$ is a strict functor. Moreover, we require it to be reduced, namely $\tilde{\A}(0)=*$ (the terminal $G$-category), and each hom-functor 
    \begin{align*}
        \tilde{\A}_{\mathbf{n}^\alpha,\mathbf{m}^\beta}:\F_G(\mathbf{n}^\alpha,\mathbf{m}^\beta)\rightarrow \Cat_G(\tilde{\A}(\mathbf{n}^\alpha),\tilde{\A}(\mathbf{m}^\beta))
    \end{align*}
    is a $G$-functor.

    \textbf{Question: In operad case, we have natural transformation between any two $G$-actions, do we need it here?}
\end{defn}

\hfil

Let $\delta_i:\mathbf{n}^\alpha\rightarrow\mathbf{1}$ be the $i$th projection (namely $i\mapsto 1$, and any $i\neq j\mapsto 0$), since it belongs to $\Pi_G$, it generates a functor $\tilde{\A}\delta_i:\tilde{\A}(\mathbf{n}^\alpha)\rightarrow \tilde{\A}(\mathbf{1})$. Hence, by universality of product, all $\tilde{\A}\delta_i$ ($1\leq i\leq n$) induces a functor $\delta:\tilde{\AA}(\mathbf{n}^\alpha)\rightarrow \tilde{A}(\mathbf{1})^n$.

\begin{defn}[$\F_{G,\vs}$ and $\F_{G,\RR}$-pseudoalgebra]
    An $\F_G$-pseudoalgebra $\tilde{\A}$ is \emph{very special} (or an $\F_{G,\vs}$-pseudoalgebra), if all $\delta:\tilde{\A}(\mathbf{n}^\alpha)\rightarrow \tilde{\A}(\mathbf{1})^{\mathbf{n}^\alpha}$ is an isomorphism for all $\mathbf{n}^\alpha\in\F_G$, and $\tilde{\A}$ is \emph{strictly special} (or an $\F_{G,\RR}$-pseudoalgebra), if all $\delta$ is in fact identity.
\end{defn}

\hfil

\begin{defn}[$\F_G$-1-cell]
    Given $\tilde{A},\tilde{B}$ two $\F_G$-pseudoalgebras, a morphism (or $\F_G$-1-cell) between them is a pseudo-transformation $(\FF,\zeta):\tilde{\A}\rightarrow \tilde{\B}$, such that restricting to $\Pi_G$, it becomes an actual natural transformation. 
\end{defn}

\begin{defn}[$\F_G$-2-cell]
    Given $\tilde{A},\tilde{B}$ two $\F_G$-pseudoalgebras, and two $\F_G$-1-cells $(\FF,\zeta),(\GG,\xi):\tilde{A}\rightarrow\tilde{\B}$, an $\F_G$-2-cell is a transformation $\lambda:\FF\Rightarrow\GG$, such that for any morphism $f:\mathbf{n}^\alpha\rightarrow\mathbf{m}^\beta$ in $\F_G$, the following two pasting diagrams are equal:
    \begin{align*}
        \xymatrix{
            \tilde{\A}(\mathbf{n}^\alpha) \ar[d]_-{\tilde{\A}f} \ar@(ur,ul)[rr]^-{\FF_{\mathbf{n}^\alpha}} \ar@{=>}[rr]^-{\lambda} \ar@(dr,dl)[rr]^-{\GG_{\mathbf{n}^\alpha}} && \tilde{\B}(\mathbf{n}^\alpha) \ar[d]^-{\tilde{\B}f} \ar@{=>}[dll]^-{\xi f}\\
            \tilde{A}(\mathbf{m}^\beta) \ar[rr]_-{\GG_{\mathbf{m}^\beta}} && \tilde{B}(\mathbf{m}^\beta)
        }\quad = \quad \xymatrix{
            \tilde{\A}(\mathbf{n}^\alpha) \ar[d]_-{\tilde{\A}f} \ar[rr]^-{\FF_{\mathbf{n}^\alpha}} && \tilde{\B}(\mathbf{n}^\alpha) \ar[d]^-{\tilde{\B}f} \ar@{=>}[dll]_-{\zeta f}\\
            \tilde{A}(\mathbf{m}^\beta) \ar@(ur,ul)[rr]_-{\FF_{\mathbf{m}^\beta}} \ar@{=>}[rr]_-{\lambda} \ar@(dr,dl)[rr]_-{\GG_{\mathbf{m}^\beta}} && \tilde{B}(\mathbf{m}^\beta)
        }
    \end{align*}
\end{defn}
