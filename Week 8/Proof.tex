\section{Abstract (ND)}

Symmetric monoidal categories are central in modern algebraic topology, especially algebraic $K$-theory relying on constructing spectra out of symmetric monoidal categories. It is known that the two inputs - the categories with action by an operad $\cP$ in the category $\Cat$ of small categories, or suitable categories described by based finite sets $\F$ - are equivalent description for symmetric monoidal categories. 

Within equivariant $K$-theory, similar machineries have been established, where the input for constructing $G$-spectra are described by an operad $\cP_G$ in the category $\Cat_G$ of small $G$-categories, or categories described by based finite $G$-sets $\F_G$. The equivalence of the two within equivariant case is a natural question to ask, and ponders the question of ``what is symmetric monoidal $G$-categories''.

We'll provide an exposition on the two inputs, and establish a 2-functor $\RR_G$ that generates $\F_G$-pseudoalgebra from $\cP_G$-pseudoalgebra, relating the two.

\section{Introduction (ND)}\label{sec_Intro}

\textbf{Fix $G$ as a finite group.} Let $P_G$ be the \emph{associativity $G$-operad} in $\Sets_G$, defined by $P_G(n):= \Sets(G,\Sigma_n)$, with structure maps defined by post composition
\begin{align*}
    \gamma:\Sets(G,\Sigma_k)\times \prod_{r=1}^k\Sets(G,\Sigma_{j_r})\cong \Sets\left(G,\Sigma_k\times \prod_{r=1}^k \Sigma_{j_r}\right)\xrightarrow{\ \gamma\circ\_\ }\Sets(G,\Sigma_j),
\end{align*},
where $j=\sum_{r=1}^k j_r$, and $\gamma:\Sigma_k\times \prod_{r=1}^k \Sigma_{j_r}\rightarrow \Sigma_j$ is the structure map for associativity operad $\mathscr{M}$ in $\Sets$. This is naturally a free $(G\times\Sigma)$-operad, as the right $(G\times \Sigma_n)$-action at each $P_G(n)$ is free.

However, it is mentioned in \cite{BBKO21} that such $G$-operad is not finitely generated, as a rough explanation, for every $n\in\NN$, there exists $m>n$ and a function $f:G\rightarrow \Sigma_m$, such that it cannot be generated using $\{P_G(k)\}_{k\leq n}$ with the operadic structure map and the $(G\times \Sigma)$-actions.

\section{Categorical Preliminaries}\label{sec_CatPrelim}
\subsection{2-Categories}\label{sub_2_cat}

\begin{defn}\label{defn_2_cat}
    A category $\C$ is a \emph{2-category}, if for each object $X,Y\in\C$ (called \emph{0-cells}), the homset $\C(X,Y)$ is itself a category (called the \emph{hom-category}). Also, for every 0-cells $X,Y,Z\in\C$, the \emph{composition} $\circ:\C(Y,Z)\times\C(X,Y)\rightarrow\C(X,Z)$ is a functor.

    Every morphism $f\in \C(X,Y)$ is called a \emph{1-cell}, while given any 1-cells $f,g\in\C(X,Y)$, a morphism $\alpha:f\Rightarrow g$ in the hom-category $\C(X,Y)$ is called a \emph{2-cell}.
\end{defn}

Here, $\circ$ must be strictly unital and associative.

\begin{rem}\label{rem_bicat}
    There is a more general notion of a \emph{bicategory}, which relaxes the composition functors to be associative and unital up to coherent natural isomorphisms, and in some literature this is called 2-categories. Here we'll take the strict associativity and unitality as axioms for 2-categories. For more general bicategory, cf. \cite{JY20}.
\end{rem}

            \ar[rr]^-{g}="2"

        && Y 
        \ar@{=>} "1";"2"^{H_1} \ar@{=>}"2";"3"^{H_2}
        }
    \end{align*}

    }
    \end{align*}
\end{defn}

\begin{align}
    (H'_2H'_1)*(H_2H_1) = (H'_2*H_2)(H'_1*H_1):(f'\circ f)\Rightarrow (h'\circ h)
\end{align}
as 2-cells.

\begin{exmp}\label{exmp_cat_of_cat}
    Let $\Cat$ denote the \emph{category of small categories}, where the objects are small categories, the hom-categories $\Cat(\A,\B)$ are functors between small categories $\A$ and $\B$, and the morphisms in $\Cat(\A,\B)$ are natural transformations between functors.

    This specifies $\Cat$ to be a 2-category. Also, a precise description of the horizontal composition of 2-cells is as follow: Given functors $F_1,F_2\in\Cat(\A,\B)$ and $G_1,G_2\in\Cat(\B,\C)$, natural transformations $\mu:F_1\Rightarrow F_2$ in $\Cat(\A,\B)$ and $\nu:G_1\Rightarrow G_2$ in $\Cat(\B,\C)$, then $\nu*\mu := \nu \circ F\mu$ defines a horizontal composition.
\end{exmp}

\subsection{Chaotic-Forgetful Adjunction}\label{sub_chaot_adj}

Given any small category $\A\in\Cat$, its object $\Ob(\A)$ forms a set; moreover, any functor $F:\A\rightarrow\B$ also has an underlying map of objects $F:\Ob(\A)\rightarrow\Ob(\B)$. Conversely, given a set, it has certain category structure:

\begin{defn}\label{defn_chaot}
    Given a set $X$, define $\mathscr{E}X$ to be the \emph{chaotic category} with underlying object set $\Ob(\mathscr{E}X):=X$, by encoding a unique isomorphism $x\xrightarrow{\sim} y$ between any element $x,y\in X$. The composition 
    \begin{align}
        (x\xrightarrow{\sim}y \xrightarrow{\sim}z) = (x\xrightarrow{\sim}z)
    \end{align}
    using the uniqueness.

    \hfil

    Given a set map $f:X\rightarrow Y$, define the functor $\mathscr{E}f:\mathscr{E}X\rightarrow\mathscr{E}Y$ so that the underlying object map is $f$, and for each morphism $x\xrightarrow{\sim}y$ in $\mathscr{E}X$, $\mathscr{E}f(x\rightarrow y):= f(x)\rightarrow f(y)$, the unique morphism between $f(x),f(y)$ in $\mathscr{E}Y$.
\end{defn}

Inspecting the definition, we see $\mathscr{E}:\Sets\rightarrow\Cat$ is a well-defined functor, called the \emph{chaotic functor}.

Notice if a functor $F:\A\rightarrow\B$ has a chaotic target $\B$, then all the morphisms $f:X\rightarrow Y$ in $\A$ must be sent to the unique morphism $F(X)\rightarrow F(Y)$ in $\B$, hence $F$ is dictated by the object map $F:\Ob(\A)\rightarrow\Ob(\B)$. Such uniqueness specifies an adjunction $(\Ob\dashv \mathscr{E})$.

\subsection{Pseudofunctor, Pseudotransform, and Modification}

\begin{defn}[Pseudofunctor]
    Given two 2-categories $\A,\B$, a \emph{pseudofunctor} $F:\A\rightarrow \B$ equips with an 0-cell map, functors between hom-categories $F_{X,Y}:\A(X,Y)\rightarrow \B(F(X),F(Y))$, an invertible 2-cell $\phi_{X,Y,Z}$ between compositions
    \begin{align*}
        \xymatrix{
            \A(Y,Z)\times \A(X,Y) \ar[d]_-{\circ} \ar[rr]^-{F_{Y,Z}\times F_{X,Y}} && \B(F(Y),F(Z))\times\B(F(X),F(Y)) \ar[d]^-{\circ} \ar@{=>}[dll]_-{\phi_{X,Y,Z}}\\
            \A(X,Z) \ar[rr]_-{F_{X,Z}} && \B(F(X),F(Z))
        }
    \end{align*}
    and another invertible 2-cell $\psi_X$ between identities
    \begin{align*}
        \xymatrix{
            F(X) \ar@/^1.5pc/[rr]^{\id_{F(X)}}="1" \ar@/_1.5pc/[rr]_{F_{X,X}(\id_X)}="2"
            \ar@{=>} "1";"2"^{\psi_X}
            && F(X)
        }
    \end{align*}
    such that the following axioms hold:
    \begin{itemize}
        \item[(i)] \textbf{(Associativity)} The following two pasting diagrams are equal:
        \begin{align*}
            \xymatrix{
                \A(Z,W)\times\A(Y,Z)\times\A(X,Y) \ar[d]_-{\circ\times\id} \ar[rr]^-{\prod F_{\_,\_}} && \B(F(Z),F(W))\times \B(F(Y),F(Z)) \times \B(F(X),F(Y)) \ar[d]^-{\circ\times \id} \ar@{=>}[dll]_-{\phi_{Y,Z,W}\times\id}\\
                \A(Y,W)\times\A(X,Y) \ar[d]_-{\circ} \ar[rr]^-{F_{Y,W}\times F_{X,Y}} && \B(F(Y),F(W))\times \B(F(X),F(Y)) \ar[d]^-{\circ} \ar@{=>}[dll]_-{\phi_{X,Y,W}}\\
                \A(X,W)\ar[rr]_-{F_{X,W}} && \B(F(X),F(W))
            }
        \end{align*}
        equals
        \begin{align*}
            \xymatrix{
                \A(Z,W)\times\A(Y,Z)\times\A(X,Y) \ar[d]_-{\id\times\circ} \ar[rr]^-{\prod F_{\_,\_}} && \B(F(Z),F(W))\times \B(F(Y),F(Z)) \times \B(F(X),F(Y)) \ar[d]^-{\id\times\circ} \ar@{=>}[dll]_-{\id\times\phi_{X,Y,Z}}\\
                \A(Z,W)\times\A(X,Z) \ar[d]_-{\circ} \ar[rr]^-{F_{Z,W}\times F_{X,Z}} && \B(F(Z),F(W))\times \B(F(X),F(Z)) \ar[d]^-{\circ} \ar@{=>}[dll]_-{\phi_{X,Z,W}}\\
                \A(X,W)\ar[rr]_-{F_{X,W}} && \B(F(X),F(W))
            }
        \end{align*}
        \item[(ii)] \textbf{(Unitality)} The following two unit diagrams commute:
        \begin{align*}
            \xymatrix{
                \A(X,Y)\times\A(X,X)\ar[rr]^-{F_{X,Y}\times F_{X,X}} && \B(F(X),F(Y))\times \B(F(X),F(X)) \ar[dd]^-{\circ} \ar@{=>}[ddll]_-{\id \mathrm{*}\psi_X}\\
                \A(X,Y)\times \mathrm{*} \ar[u]^-{\id\times \id_X} \ar[d]_-{F_{X,Y}\times \id_{F(X)}}\\
                \B(F(X),F(Y))\times \B(F(X),F(X)) \ar[rr]_-{\circ} && \B(F(X),F(Y))
            }
        \end{align*}
        \begin{align*}
            \xymatrix{
                \A(Y,Y)\times\A(X,Y)\ar[rr]^-{F_{Y,Y}\times F_{X,Y}} && \B(F(Y),F(Y))\times \B(F(X),F(Y)) \ar[dd]^-{\circ} \ar@{=>}[ddll]_-{\psi_Y\mathrm{*}\id}\\
                \mathrm{*}\times\A(X,Y) \ar[u]^-{\id_Y\times \id} \ar[d]_-{\id_{F(Y)}\times F_{X,Y}}\\
                \B(F(Y),F(Y))\times \B(F(X),F(Y)) \ar[rr]_-{\circ} && \B(F(X),F(Y))
            }
        \end{align*}
        where $\id_X:*\rightarrow\A(X,X)$ is the functor indicating identity of $X$.
    \end{itemize}

    If $\psi_X$ is identity 2-cell for all 0-cell $X$, then the pseudofunctor $F$ is called \emph{normal}.
\end{defn}

\hfil

\begin{defn}[Pseudotransform]
    Fix 2-categories $\A,\B$, let $(F,\phi,\psi), (G,\varphi,\vartheta):\A\rightarrow\B$ be two pseudofunctors. A \emph{pseudotransform} $\mu:(F,\phi,\psi)\rightarrow (G,\varphi,\vartheta)$ collects 1-cells $\mu_X:F(X)\rightarrow G(X)$ for each 0-cell $X\in\A$, and for each 1-cell $f:X\rightarrow Y$ in $\A$, associates a 2-cell $\mu f$ such that 
    \begin{align*}
        \xymatrix{
            F(X) \ar[r]^-{\mu_X} \ar[d]_-{Ff} & G(X) \ar[d]^-{Gf} \ar@{=>}[dl]_-{\mu f}\\
            F(Y) \ar[r]_-{\mu_Y} & G(Y)
        }
    \end{align*}
    commutes. 

    Furthermore, they satisfy the following axioms:
    \begin{itemize}
        \item[(i)] (\textbf{Associativity}) Let $f:X\rightarrow Y$, $g:Y\rightarrow Z$ be 1-cells in $\A$, the following two pasting diagrams are equal:
        \begin{align*}
            \xymatrix{
                F(X) \ar[dd]_-{F(g\circ f)} \ar[dr]^-{Ff} \ar[rr]^-{\mu_X} && G(X) \ar@{=>}[dl]_-{\mu f} \ar[dr]^-{Gf}\\
                & F(Y) \ar[rr]^-{\mu_Y} \ar@{=>}[l]_-{\phi_{g,f}} \ar[dl]^-{Fg} && G(Y) \ar[dl]^-{Gg} \ar@{=>}[dlll]_-{\mu g}\\
                F(Z) \ar[rr]_-{\mu_Z} && G(Z)
            } \quad = \quad \xymatrix{
                F(X) \ar[dd]_-{F(g\circ f)} \ar[rr]^-{\mu_X} && G(X) \ar[dr]^-{Gf} \ar[dd]_-{G(g\circ f)} \ar@{=>}[ddll]_-{\mu(g\circ f)}\\
                &&& G(Y) \ar[dl]^-{Gg} \ar@{=>}[l]_-{\varphi_{g,f}}\\
                F(Z) \ar[rr]_-{\mu_Z} && G(Z)
            }
        \end{align*}
        \item[(ii)] (\textbf{Unitality}) The following pasting diagrams are equal:
        \begin{align*}
            \xymatrix{
                F(X) \ar@(dl,ul)[dd]_-{F(\id_X)} \ar@(dr,ur)[dd]^-{\id_{F(X)}} \ar@{=>}[dd]_-{\psi_X} \ar[rr]^-{\mu_X} && G(X) \ar[dd]^-{\id_{G(X)}}\\
                \\
                F(X) \ar[rr]_-{\mu_X} && G(X)
            }\quad = \quad \xymatrix{
                F(X) \ar[dd]_-{F(\id_X)} \ar[rr]^-{\mu_X} && G(X) \ar@{=>}[ddll]_-{\mu(\id_X)} \ar@(dl,ul)[dd]_-{G(\id_X)} \ar@(dr,ur)[dd]^-{\id_{G(X)}} \ar@{=>}[dd]^-{\vartheta_X}\\
                \\
                F(X) \ar[rr]_-{\mu_X} && G(X)
            }
        \end{align*}
    \end{itemize}
\end{defn}

\hfil

\begin{defn}[Modification]
    Continue with the same notation as above. Let $\mu,\nu:(F,\phi,\psi)\rightarrow (G,\varphi,\vartheta)$ be two pseudotransforms, a \emph{modification} $\lambda:\mu\Rightarrow\nu$ is a collection of 2-cells $\lambda_X:\mu_X\Rightarrow\nu_X$ for each 0-cell $X\in\A$, such that the following two pasting diagrams are equal, for each 1-cell $f:X\rightarrow Y$ in $\A$:
    \begin{align*}
        \xymatrix{
            F(X) \ar[dd]_-{Ff} \ar@(ur,ul)[rr]^-{\mu_X} \ar@(dr,dl)[rr]_-{\nu_X} \ar@{=>}[rr]^-{\lambda_X} && G(X) \ar@{=>}[ddll]^-{\nu f} \ar[dd]^-{Gf}\\
            \\
            F(Y) \ar[rr]_-{\nu_Y} && G(Y)
        }\quad = \quad \xymatrix{
            F(X) \ar[dd]_-{Ff} \ar[rr]^-{\mu_X} && G(X) \ar[dd]^-{Gf} \ar@{=>}[ddll]_-{\mu f}\\
            \\
            F(Y) \ar@(ur,ul)[rr]^-{\mu_Y} \ar@{=>}[rr]^-{\lambda_Y} \ar@(dr,dl)[rr]_-{\nu_Y} && G(Y)
        }
    \end{align*}

\end{defn}

\subsection{$G$-Categories}

\begin{defn}
    A small category $\A$ is a \emph{$G$-category}, if the object and morphism sets $\Ob(\A), \Mor(\A)$ are both (left) $G$-sets, such that the following four structure maps $(S,T,I,C)$ (stands for \emph{source, target, identity, and composition}) are all $G$-equivariant:
    \begin{itemize}
        \item $S,T:\Mor(\A)\rightarrow\Ob(\A)$ by $S(X\xrightarrow{f}Y):= X$, and $T(X\xrightarrow{f}Y):= Y$.
        \item $I:\Ob(\A)\rightarrow\Mor(\A)$ by $I(X):= \id_X$.
        \item $C:\Mor(\A)\times_{S,T}\Mor(\A)\rightarrow\Mor(\A)$ by $C(g,f):= g\circ f$.
    \end{itemize}
    Here, $\Mor(\A)\times_{S,T}\Mor(\A):= \{(g,f)\in \Mor(\A)^2\ |\ S(g)=T(f)\}$, precisely the pair of morphisms that're composable, and it endows $G$-set structure via restricting the diagonal $G$-action on $\Mor(\A)^2$. This is well-defined due to the assumption $S,T$ are $G$-maps.
\end{defn}

Another equivalent characterization can be viewed as a group homomorphism $G\rightarrow\Aut_{\Cat}(\A)$, which assigns each $g\in G$ an automorphism functor on $\A$ such that composition respects $G$'s multiplication structure, and identity gets sent to the identity functor. So, from now on we view each element $g\in G$ as a functor, whenever a category is specified as a $G$-category.

\hfil

\begin{defn}
Let $\Cat_G$ denotes the 2-category of small $G$-categories, with (nonequivariant) functors as $1$-cells, natural transformations as $2$-cells. The hom-category between any two $G$-categories $\Cat_G(\A,\B)$ also has a natural $G$-action via conjugation.  Which, the $G$-fixed subcategory of the hom-category $[\Cat_G(\A,\B)]^G$ precisely describes all $G$-functor, or the functors $F:\A\rightarrow\B$ such that $g\circ F=F\circ g$ for all $g\in G$.
\end{defn}

\hfil

Another important notion is the twisted $G$-action on products when given a group homomorphism $\alpha:G\rightarrow\Sigma_n$. Given a category $\A$, the $n$-fold product $\A^\mathbf{n}$ has a natural left $\Sigma_n$-action via permuting the coordinates; if $\A$ is a $G$-category, $\A^\mathbf{n}$ also endows a natural diagonal $G$-action. The two coordinates to give a natural left $(G\times \Sigma_n)$-action on $\A^\mathbf{n}$.

\begin{defn}
    Let $\A^{\mathbf{n}^\alpha}$ be a $G$-category with the underlying category being $\A^\mathbf{n}$, with the $G$-action defined as follow:
    \begin{align*}
        g\cdot_\alpha (X_1,...,X_n):= (g\cdot X_{\alpha(g)^{-1}(1)},...,g\cdot X_{\alpha(g)^{-1}(n)})
    \end{align*}
\end{defn}
Another characterization can be understood as follow: Given the graph subgroup $\Gamma_\alpha:= \{(g,\alpha(g))\in G\times\Sigma_n\}_{g\in G}$, the projection $\pi_G:G\times\Sigma_n\twoheadrightarrow G$ restricts to an isomorphism $\Gamma_\alpha\xrightarrow{\sim}G$. The $\alpha$-twisted action is induced by the pullback along this isomorphism.

\section{Categorical Preliminaries: Operads (Consider cutting this section, and preserve the operad $\cP$ only)}
\subsection{Symmetric Monoidal Categories}

\begin{defn}
    A category $\mathcal{V}$ is a \emph{monoidal category}, if there is a bifunctor $\tensor:\mathcal{V}\times\mathcal{V}\rightarrow\mathcal{V}$ (called \emph{monoidal product}), and an object $\kappa\in\mathcal{V}$ (called \emph{unit/monoidal unit}), together with the following natural isomorphisms:
    \begin{itemize}
        \item The \emph{associator} $a:(\_\tensor \_)\tensor \_ \xrightarrow{\sim} \_\tensor(\_\tensor \_)$ of functors $\mathcal{V}^3\xrightarrow{\sim}\mathcal{V}$.
        \item The \emph{left unit} $\lambda:\kappa\tensor\_\xrightarrow{\sim} \Id_{\mathcal{V}}$ and the \emph{right unit} $\rho:\_\tensor \kappa\xrightarrow{\sim}\Id_\mathcal{V}$ of functors $\V\rightarrow\V$.
        \item The \emph{Symmetric Braiding} $B:\_ _1\tensor\_ _2\xrightarrow{\sim}\_ _2\tensor\_ _1$, with extra property $B_{Y,X}:Y\tensor X\rightarrow X\tensor Y$ equals to $B_{X,Y}^{-1}:X\tensor Y\rightarrow Y\tensor X$.
    \end{itemize} 

    They also satisfy the following axioms:
    \begin{itemize}
        \item \emph{Triangle Axiom:}
        \[\xymatrix{
            (X\tensor \kappa)\tensor Y \ar[dr]_-{\rho_X \tensor \id_Y} \ar[rr]^-{a_{X,\kappa,Y}} && X\tensor (\kappa\tensor Y) \ar[dl]^-{\id_X\tensor \lambda_Y}\\
            & X\tensor Y
        }\]
        \item \emph{Pentagon Axiom:}
        \[\xymatrix{
            ((X\tensor Y)\tensor Z)\tensor W \ar[rr]^-{a_{X\tensor Y,Z,W}} \ar[d]_-{a_{X,Y,Z}\tensor \id_W} && (X\tensor Y)\tensor(Z\tensor W) \ar[d]^-{a_{X,Y,Z\tensor W}}\\
            (X\tensor (Y\tensor Z))\tensor W \ar[dr]_-{a_{X,Y\tensor Z,W}} && X\tensor (Y\tensor (Z\tensor W)) \\
            & X\tensor ((Y\tensor Z)\tensor W) \ar[ur]_-{\id_X\tensor a_{Y,Z,W}}
        }\]
        \item \emph{Hexagon Axiom:}
        \[\xymatrix{
            (X\tensor Y)\tensor Z \ar[r]^-{a_{X,Y,Z}} \ar[d]_-{B_{X,Y}\tensor \Id_Z} & X\tensor (Y\tensor Z)\ar[r]^-{B_{X,Y\tensor Z}} & (Y\tensor Z)\tensor X \ar[d]^-{a_{Y,Z,X}}\\
            (Y\tensor X)\tensor Z \ar[r]_-{a_{Y,X,Z}} & Y\tensor (X\tensor Z) \ar[r]_-{\id_Y\tensor B_{X,Z}} & Y\tensor (Z\tensor X)
        }\]
        \[\xymatrix{
            X\tensor (Y\tensor Z) \ar[d]_-{\id_X\tensor B_{Y,Z}} & (X\tensor Y)\tensor Z \ar[l]_-{a_{X,Y,Z}} \ar[r]^-{B_{X\tensor Y,Z}} & Z\tensor (X\tensor Y)\\
            X\tensor (Z\tensor Y) & (X\tensor Z)\tensor Y \ar[l]^-{a_{X,Z,Y}} \ar[r]_-{B_{X,Z}\tensor \id_Y} & (Z\tensor X)\tensor Y \ar[u]_-{a_{Z,X,Y}}
        }\]
    \end{itemize}
\end{defn}
Based on \emph{Maclane's Coherence Theorem}, the above two axioms guarantees all finite sequences of $a,\lambda,\rho$ and their inverses agree, namely there is a unique isomorphism $P_1\rightarrow P_2$, where $P_i$ are (possibly) different orders of performing product on $X_1,...,X_n\in\mathcal{V}$. For details, cf. \cite{M98}.

\subsection{Operads, Operadic Algebras}

\begin{defn}
Given symmetric monoidal category $\mathcal{V}$, a \emph{$\Sigma$ operad} $\mathcal{C}$ in $\mathcal{V}$ is a collection of objects $\{\mathcal{C}(j)\in\mathcal{V}\}_{j\in\NN}$, each $\mathcal{C}(j)$ with a right $\Sigma_j$-action (a group homomorphism $\Sigma_j^\Op\rightarrow \Aut_\mathcal{V}(\mathcal{C}(j))$), a \emph{unit map} $\eta:1\rightarrow \mathcal{C}(1)$, together with composition maps
\[\gamma:\mathcal{C}(k)\tensor \mathcal{C}(j_1)\tensor...\tensor \mathcal{C}(j_k)\rightarrow \mathcal{C}(j_1+...+j_k)\]
that satisfies the following axioms:
\begin{itemize}
    \item \emph{Unit:}
    \[\xymatrix{
        \mathcal{C}(j)\tensor \kappa^{\tensor j} \ar[dr]_-{\id_{\mathcal{C}(j)}\tensor \eta^{\tensor j}} \ar[rr]^-{\sim} && \mathcal{C}(j)\\
        & \mathcal{C}(j)\tensor \mathcal{C}(1)^{\tensor j} \ar[ur]_-{\gamma}
    },\quad\quad \xymatrix{
        \kappa\tensor \mathcal{C}(j) \ar[dr]_-{\eta\tensor \id_{\mathcal{C}(j)}} \ar[rr]^-{\sim} && \mathcal{C}(j)\\
        & \mathcal{C}(1)\tensor \mathcal{C}(j) \ar[ur]_-{\gamma}
    }\]
    \item \emph{Associativity:} Fix natural numbers $k$, $j = \sum_{s=1}^k j_s$, define index $g_s:= j_1+...+j_s$ for $1\leq s\leq k$. For each $i_1,...,i_{g_k}=i_j$, with sum $i=\sum_{r=1}^j i_r$, and $h_s:= i_{g_{(s-1)}+1}+...+i_{g_s}$ (so $i=\sum_{s=1}^k h_s$ also), the following diagram commutes:
    \[\xymatrix{
        \left[\mathcal{C}(k)\tensor \left(\bigtensor_{s=1}^k \mathcal{C}(j_s)\right)\right]\tensor \left(\bigtensor_{r=1}^j \mathcal{C}(i_r)\right) \ar[dd]_-{\text{shuffle}} \ar[rr]^-{\gamma\tensor \id} && \mathcal{C}(j) \tensor \left(\bigtensor_{r=1}^j \mathcal{C}(i_r)\right) \ar[d]^-{\gamma}\\
        && \mathcal{C}(i)\\
        \mathcal{C}(k) \tensor \left[\bigtensor_{s=1}^k \left(\mathcal{C}(j_s)\tensor \left(\bigtensor_{q=1}^{j_s} \mathcal{C}(i_{g_{(s-1)}+q})\right)\right)\right] \ar[rr]_-{\id_{\mathcal{C}(k)} \tensor (\bigtensor_s \gamma)} && \mathcal{C}(k)\tensor \left(\bigtensor_{s=1}^k \mathcal{C}(h_s)\right) \ar[u]_-{\gamma}
    }\]
    \item \emph{Equivariance:}
    \begin{itemize}
        \item Given $\sigma\in\Sigma_k$, $j_1+...+j_k = \Sigma_j$, let $\sigma(j_1,...,j_k)\in\Sigma_j$ denotes ``permutation of $k$-blocks of letters'', by $(j_1,...,j_k)\mapsto (j_{\sigma^{-1}(1)},...,j_{\sigma^{-1}(k)})$.
        \begin{exmp}
            Given $j=5=2+3$, the permutation $(1\ 2)\in \Sigma_2$, $\sigma(2,3)$ is the permutation $\langle 1,2;3,4,5 \rangle \mapsto \langle 3,4,5;1,2\rangle$ (namely changing the blocks' order, without changing the order in each block).
        \end{exmp} 
        Then, the following diagram commutes:
        \[\xymatrix{
            \mathcal{C}(k)\tensor \left(\bigtensor_{r=1}^k \mathcal{C}(j_r)\right) \ar[d]_-{\gamma} \ar[rr]^-{\sigma\tensor \text{ shuffle}_{\sigma^{-1}}} && \mathcal{C}(k)\tensor \left(\bigtensor_{r=1}^k \mathcal{C}(j_{\sigma(r)})\right) \ar[d]^-{\gamma}\\
            \mathcal{C}(j) \ar[rr]_-{\sigma(j_{\sigma(1)},...,j_{\sigma(k)})} && \mathcal{C}(j)
        }\]
        \item Given $\tau_r\in \Sigma_{j_r}$, denote $\tau_1\oplus ... \oplus \tau_k\in \Sigma_j$ by letting each $\tau_r$ permuting the $j_r$th block in $j$ (for instance, let $\tau_1$ acts on $1,...,j_1$; let $\tau_2$ acts on $j_1+1,...,j_1+j_2$, etc.). Then, the following diagram commutes:
        \[\xymatrix{
            \mathcal{C}(k)\tensor \left(\bigtensor_{r=1}^k \mathcal{C}(j_r)\right)\ar[d]_-{\gamma} \ar[rr]^-{\id_{\mathcal{C}(k)}\tensor (\bigtensor_r \tau_r)} && \mathcal{C}(k)\tensor \left(\bigtensor_{r=1}^k \mathcal{C}(j_r)\right) \ar[d]^-{\gamma}\\
            \mathcal{C}(j) \ar[rr]_{\tau_1\oplus...\oplus\tau_k} && \mathcal{C}(j)
        }\]
    \end{itemize}
\end{itemize}

If $\mathcal{C}(0)=\kappa$ (with $\gamma:\mathcal{C}(0)\rightarrow \mathcal{C}$ being $\id_{\mathcal{C}(0)}$), the operad is \emph{reduced}.
\end{defn}

\begin{defn}
    Let $\mathcal{C},\mathcal{C}'$ be two operads on symmetric monoidal category $\mathcal{V}$, a \emph{morphism of operads} $\psi:\mathcal{C}\rightarrow \mathcal{C}'$, is a sequence of $\Sigma_j$-equivariant morphisms $\psi_j:\mathcal{C}(j)\rightarrow \mathcal{C}'(j)$ in $\mathcal{V}$ (i.e., $\sigma^{-1}\circ \psi_j\circ \sigma = \psi_j$, for all $\sigma\in \Sigma_j$), such that it preserves the operad's structure morphism:
    \[\xymatrix{
        \mathcal{C}(k)\tensor \mathcal{C}(j_1)\tensor...\tensor \mathcal{C}(j_k) \ar[d]_-{\psi_k\tensor \psi_{j_1}\tensor...\tensor \psi_{j_k}} \ar[r]^-{\gamma} & \mathcal{C}(j_1+...+j_k) \ar[d]^-{\psi_{j_1+...+j_k}}\\
        \mathcal{C}'(k)\tensor \mathcal{C}'(j_1)\tensor ..\tensor \mathcal{C}'(j_k) \ar[r]_-{\gamma'} & \mathcal{C}'(j_1+...+j_k)
    }\]
\end{defn}

\hfil

\begin{defn}
    Given an operad $\mathcal{C}$ over a symmetric monoidal category $\mathcal{V}$, a \emph{$\mathcal{C}$-algebra} is an object $A\in\mathcal{V}$, with collections of morphisms $\theta:\mathcal{C}(j)\tensor A^j\rightarrow A$, that satisfies the following axioms:
    \begin{itemize}
        \item \emph{Associativity:} Given $j = \sum_{i=1}^k j_i$, the following diagram commutes:
        \[\xymatrix{
            \left[C(k)\tensor \left(\bigtensor_{i=1}^k C(j_i)\right)\right] \tensor A^{\tensor j} \ar[dd]_-{\text{shuffle}} \ar[rr]^-{\gamma\tensor \id_{A^{\tensor j}}} && C(j)\tensor A^{\tensor j} \ar[dr]^-{\theta}\\
            & & & A\\
            C(k)\tensor \left[\bigtensor_{i=1}^k \left(C(j_i)\tensor A^{\tensor j_i}\right)\right] \ar[rr]_-{\id_{C(k)}\tensor \theta^k} && C(k)\tensor A^{\tensor k} \ar[ur]_-{\theta}
        }\]
        (Note: This part uses the isomorphism $A^{\tensor j}\cong\bigtensor_{i=1}^k A^{\tensor j_i})$.
        \item \emph{Unit:} The following diagram commutes:
        \[\xymatrix{
            \kappa \tensor A \ar[dr]_-{\eta\tensor \id_A} \ar[rr]^-{\sim} && A\\
            & C(1)\tensor A \ar[ur]_-{\theta}
        }\]
        \item \emph{Eauivariance:} Given any permutation $\sigma\in\Sigma_j$, the following diagram commutes:
        \[\xymatrix{
            C(j)\tensor A^{\tensor j} \ar[dr]_-{\theta} \ar[rr]^-{\sigma\tensor \sigma^{-1}} && C(j)\tensor A^{\tensor j} \ar[dl]^-{\theta}\\
            & A
        }\]
    \end{itemize}
\end{defn}

\subsection{Permutativity Operad $\cP$}
We first starts by introducing a relevant operad in $\Sets$: The \emph{associativity operad} $\M$ in $\Sets$ is defined as $\M(j):=\Sigma_j$ (the symmetric group for cardinality $j$), which endows a natural right $\Sigma_j$-action via right multiplication. The operadic structure map is given by 
\begin{align}
    \gamma:\Sigma_k\times\prod_{r=1}^k\Sigma_{j_r}\rightarrow\Sigma_j,\quad \gamma(\sigma;\tau_1,...,\tau_k):= \tau_{\sigma^{-1}(1)}\oplus...\oplus\tau_{\sigma^{-1}(k)}\circ \sigma(j_{\sigma(1)},...,j_{\sigma(k)})
\end{align}
where $j=\sum_r j_r$, $\sigma(j_1,...,j_k)\in \Sigma_j$ denotes the block permutation of $j_1,...,j_k$ within $j$ (in order), and $\tau_1\oplus ... \tau_{k}$ is the image of the embedding $\oplus:\prod_{r=1}^k\Sigma_{j_r}\rightarrow\Sigma_j$, by viewing the $r$th block as finite set of cardinality $j_r$. Inspecting the definitions, the algebra over $\M$ precisely describes the set theoretic monoids. More details can be found in \textbf{(Yet to be determined...)}

\hfil

Now, recall that the chaotic functor $\mathscr{E}:\Sets\rightarrow\Cat$ is a right adjoint of $\Ob:\Cat\rightarrow\Sets$, in particular it preserves products. Hence, the chaotic categorification $\cP(j):= \mathscr{E}\M(j)$ also has a natural operadic structure, via the identification
\begin{align}
    \mathscr{E}\M(k)\times \prod_{r=1}^k \mathscr{E}\M(j_r)\cong \mathscr{E}\left(\M(k)\times\prod_{r=1}^k\M(j_r)\right) \xrightarrow{\mathscr{E}\gamma}\mathscr{E}\M(j).
\end{align}
This is called the \emph{permutativity operad in $\Cat$}, or the \emph{Barratt-Eccles operad} as credited to the authors who proposed it. As an abuse of notation, the operadic structure map of $\cP$ is also denoted $\gamma$.

\hfil

It is known in \textbf{(Yet to be determined...check sym mon cat)} that the algebras over $\cP$ is isomorphic to the permutative categories. Later on, in \textbf{(Yet to be determined...)} the concept of pseudoalgebra over $\cP$ is proposed, and in \cite{LMRZ26} it specified an equivalence between the $\cP$-pseudoalgebras and the symmetric monoidal categories, by relaxing the strict associativity/unity in $\cP$-algebras to be only up to coherent natural isomorphisms.

    Let $(\FF,\zeta),(\GG,\xi):(\A,\theta,\phi)\rightarrow (\B,\vartheta,\varphi)$ be two $\cP_G$-pseudomorphisms, a $\cP_G$-2-cell $\lambda:(\FF,\zeta)\Rightarrow(\GG,\xi)$ is a natural transformation $\lambda:\FF\rightarrow\GG$, such that the following two pasting diagrams are equal:
    \begin{align*}
        \xymatrix{
            \cP_G(j)\times \A^j \ar@(ur,ul)[r]^-{\id\times \FF^j} \ar@{=>}[r]^-{\id\times \lambda^j} \ar@(dr,dl)[r]^-{\id\times\GG^j} \ar[d]_-{\theta_j} & \cP_G(j)\times \B^j \ar[d]^-{\vartheta_j} \ar@{=>}[dl]^-{\xi_j}\\
            \A \ar[r]_-{\GG} & \B
        }\quad = \quad \xymatrix{
            \cP_G(j)\times \A^j \ar[d]_-{\theta_j} \ar[r]^-{\id\times \FF^j} & \cP_G(j)\times\B^j \ar[d]^-{\vartheta_j} \ar@{=>}[dl]_-{\zeta_j}\\
            \A \ar@(ur,ul)[r]_-{\FF} \ar@{=>}[r]_-{\lambda} \ar@(dr,dl)[r]_-{\GG} & \B
        }
    \end{align*}
\end{defn}

\subsection{Category $\F_G$}

\begin{defn}
    $\F_G$ denotes the \emph{category of based finite sets}, where the objects are $\mathbf{n}^\alpha$, set wise being the finite set $\{0,1,...,n\}$ with $0$ fixed as the basepoint, and $\alpha:G\rightarrow\Sigma_n$ is a based $G$-action on the set.

    The homset $\F_G(\mathbf{n}^\alpha,\mathbf{m}^\beta)$ consists of all the based set map, which the homset also endows with a $G$-action via conjugation. From now on, we'll view $\F_G$ as a 2-category, where each hom-category $\F_G(\mathbf{n}^\alpha,\mathbf{m}^\beta)$ is discrete with only identity 2-cells.

    \hfil

    Furthermore, we define the following:
    \begin{itemize}
        \item $\Pi_G\subset \F_G$ denotes the subcategory where each morphism $\phi:\mathbf{n}^\alpha\rightarrow\mathbf{m}^\beta$ satisfies $|\phi^{-1}(j)| = 0,1$ for all $j\neq 0$.
        \item $\Lambda_G,\Lambda_G^\perp,\Sigma_G\subset\Pi_G$ denotes the subcategory where each morphism is injective, surjective, and bijective respectively (in particular, $\Lambda_G\cap \Lambda_G^\perp = \Sigma_G$).
        \item $\mathscr{E}_G\subset \F_G$ denotes the subcategory where each morphism satisfies $\phi^{-1}(0)=\{0\}$.
    \end{itemize}
\end{defn}

\hfil

\subsection{2-Category $\F_G$-$\PsAlg$}

\begin{defn}[$\F_G$-pseudoalgebra]
    An $\F_G$-pseudoalgebra, is a pseudofunctor $\tilde{\A}:\F_G\rightarrow \Cat_G$, such that restricting to the subcategory $\Pi_G\subset\F_G$, $\tilde{\A}$ is a strict functor. Moreover, we require it to be reduced, namely $\tilde{\A}(0)=*$ (the terminal $G$-category), and each hom-functor 
    \begin{align*}
        \tilde{\A}_{\mathbf{n}^\alpha,\mathbf{m}^\beta}:\F_G(\mathbf{n}^\alpha,\mathbf{m}^\beta)\rightarrow \Cat_G(\tilde{\A}(\mathbf{n}^\alpha),\tilde{\A}(\mathbf{m}^\beta))
    \end{align*}
    is a $G$-functor.
\end{defn}

\hfil

Let $\delta_i:\mathbf{n}^\alpha\rightarrow\mathbf{1}$ be the $i$th projection (namely $i\mapsto 1$, and any $i\neq j\mapsto 0$), since it belongs to $\Pi_G$, it generates a functor $\tilde{\A}\delta_i:\tilde{\A}(\mathbf{n}^\alpha)\rightarrow \tilde{\A}(\mathbf{1})$. Hence, by universality of product, all $\tilde{\A}\delta_i$ ($1\leq i\leq n$) induces a functor $\delta:\tilde{\A}(\mathbf{n}^\alpha)\rightarrow \tilde{\A}(\mathbf{1})^{\mathbf{n}^\alpha}$.

\begin{defn}[$\F_{G,\vs}$ and $\F_{G,\RR}$-pseudoalgebra]
    An $\F_G$-pseudoalgebra $\tilde{\A}$ is \emph{very special} (or an $\F_{G,\vs}$-pseudoalgebra), if $\delta:\tilde{\A}(\mathbf{n}^\alpha)\rightarrow \tilde{\A}(\mathbf{1})^{\mathbf{n}^\alpha}$ is an isomorphism for all $\mathbf{n}^\alpha\in\F_G$, and $\tilde{\A}$ is \emph{strictly special} (or an $\F_{G,\RR}$-pseudoalgebra), if all $\delta$ is in fact identity.
\end{defn}

\hfil

\begin{defn}[$\F_G$-1-cell]
    Given $\tilde{A},\tilde{B}$ two $\F_G$-pseudoalgebras, a morphism (or $\F_G$-1-cell) between them is a pseudo-transformation $(\FF,\zeta):\tilde{\A}\rightarrow \tilde{\B}$, such that restricting to $\Pi_G$, it becomes an actual natural transformation. Moreover, it respects $G$-equivariance for every $g\in G$ and morphism $f\in\Mor(\F_G)$:
    \begin{align*}
        \zeta(g\circ f\circ g^{-1}) = g\circ \zeta(f)\circ g^{-1}
    \end{align*}
\end{defn}

\hfil

\begin{defn}[$\F_G$-2-cell]
    Given $\tilde{A},\tilde{B}$ two $\F_G$-pseudoalgebras, and two $\F_G$-1-cells $(\FF,\zeta),(\GG,\xi):\tilde{A}\rightarrow\tilde{\B}$, an $\F_G$-2-cell is a transformation $\lambda:\FF\Rightarrow\GG$, such that for any morphism $f:\mathbf{n}^\alpha\rightarrow\mathbf{m}^\beta$ in $\F_G$, the following two pasting diagrams are equal:
    \begin{align*}
        \xymatrix{
            \tilde{\A}(\mathbf{n}^\alpha) \ar[d]_-{\tilde{\A}f} \ar@(ur,ul)[rr]^-{\FF_{\mathbf{n}^\alpha}} \ar@{=>}[rr]^-{\lambda} \ar@(dr,dl)[rr]^-{\GG_{\mathbf{n}^\alpha}} && \tilde{\B}(\mathbf{n}^\alpha) \ar[d]^-{\tilde{\B}f} \ar@{=>}[dll]^-{\xi f}\\
            \tilde{A}(\mathbf{m}^\beta) \ar[rr]_-{\GG_{\mathbf{m}^\beta}} && \tilde{B}(\mathbf{m}^\beta)
        }\quad = \quad \xymatrix{
            \tilde{\A}(\mathbf{n}^\alpha) \ar[d]_-{\tilde{\A}f} \ar[rr]^-{\FF_{\mathbf{n}^\alpha}} && \tilde{\B}(\mathbf{n}^\alpha) \ar[d]^-{\tilde{\B}f} \ar@{=>}[dll]_-{\zeta f}\\
            \tilde{A}(\mathbf{m}^\beta) \ar@(ur,ul)[rr]_-{\FF_{\mathbf{m}^\beta}} \ar@{=>}[rr]_-{\lambda} \ar@(dr,dl)[rr]_-{\GG_{\mathbf{m}^\beta}} && \tilde{B}(\mathbf{m}^\beta)
        }
    \end{align*}
\end{defn}

\section{Counterexample: $\F_{G,\RR}$-pseudoalgebra that's not $\cP_G$-pseudoalgebra}

Here we fix the group $G:=\ZZ/2\ZZ$. Which, we have $\cP_G(1)=\cP_G(1) = *$, in particular $G\times \Sigma_1\cong G$ has trivial action on these. This leads to the following properties of $\cP_G$ or $\cP_G$-pseudoalgebra:

\begin{prop}
    For any $\cP_G$-pseudoalgebra $\A$, any $G$-action $g:\A\xrightarrow{\sim} \A$ is naturally isomorphic to the identity functor $\Id_\A$.
\end{prop}
\begin{proof}
    Pasting the unitality and equivariance diagram together, we get the following naturality diagram:
    \begin{align}
        \xymatrix{
            \A \ar@(dl,l)[ddr]_-{\Id_\A} \ar[dr]^-{\iota} \ar[rrrr]^-{g^{-1}} &&&& \A \ar[dl]_-{\iota} \ar@(dr,r)[ddl]^-{\Id_\A}\\
            &\cP_G(1)\times\A \ar@{=>}[l]^-{\phi_1} \ar[d]_-{\theta_1} \ar[rr]^-{g\times g^{-1}} && \cP_G(1)\times\A \ar[d]^-{\theta_1} \ar@{<=}[r]_-{\phi_1^{-1}} & \\
            &\A \ar@{=}[rr] && \A
        }
    \end{align}
    Here, the top trapezoid commutes because $G$ acts trivially on $\cP_G(1)$. This identifies a natural isomorphism as follow:
    \begin{align}
        g^{-1}=\Id_\A\circ g^{-1} \xRightarrow{\phi_1^{-1}*\id_{g^{-1}}} (\theta_1\circ\iota)\circ g^{-1} = \theta_1\circ (g\times g^{-1})\circ \iota = \theta_1\circ \iota \xRightarrow{\phi_1} \Id_\A
    \end{align}
\end{proof}
%Self note: In particular, this extends the $G$-action $G\rightarrow \Aut_{\Cat_G}(\A)$ into a functor $\mathscr{E}G\rightarrow \Aut_{\Cat_G}(\A)$.

\hfil

To construct a counterexample for our fixed $G$, it suffices to find an $\F_{G,\RR}$-pseudoalgebra that doesn't have the above property. Here, we largely use the fact that for two naturally isomorphic functors, if the target category is discrete, the two functors must be equal.

\begin{con}
Let $\A:=\{0,1\}$ denotes the discrete category with $2$ objects, which a $G$-action on $\A$ is specified by its action on the objects due to its discreteness. Define a $G$-action on $\A$ given by 
\begin{align}
    &H_0:\A\xrightarrow{\sim}\A,\quad H_0(0)=0,\ H_0(1)=1\\
    &H_1:\A\xrightarrow{\sim}\A,\quad H_1(0)=1,\ H_1(1)=0,
\end{align}
Which, $H_0^2=H_0$, $H_0\circ H_1=H_1\circ H_0$, and $H_1^2=H_0$, so $H_0,H_1$ satisfies the relations of $G=\ZZ/2\ZZ$, hence inducing a $G$-action on $\A$.

\hfil

Define $\tilde{\A}:\F_G\rightarrow \Cat_G$ given by $\tilde{\A}(\mathbf{n}^\alpha):=\A^{\mathbf{n}^\alpha}$ (with underlying category $\A^n$, a discrete category with $2^n$ objects), and for any morphism $f:\mathbf{n}^\alpha\rightarrow\mathbf{m}^\beta$, define $\tilde{\A}f:\A^{\mathbf{n}^\alpha}\rightarrow\A^{\mathbf{m}^\beta}$ by 
\begin{align}
    \tilde{\A}f\left((a_i)_{i=1,...,n}\right):= \left(\sum_{i\in f^{-1}(j)}a_i\ \mathrm{mod}\ 2 \right)_{j=1,...,m},
\end{align}
for special case where $f^{-1}(j)=\emptyset$, define the $j$th entry as $0$.

To check it preserves composition up to coherent natural isomorphism, consider $\mathbf{n}^\alpha\xrightarrow{f}\mathbf{m}^\beta\xrightarrow{g}\mathbf{k}^\gamma$ in $\F_G$, then $\tilde{\A}(g\circ f):\tilde{\A}^{\mathbf{n}^\alpha}\rightarrow\tilde{\A}^{\mathbf{m}^\beta}$ is given as follow:
\begin{align}
    \tilde{\A}(g\circ f)((a_i)_{i=1,...,n}) &= \left(\sum_{i\in (g\circ f)^{-1}(\ell)}a_i\ \mathrm{mod}\ 2\right)_{\ell=1,...,k}\\
    &= \left(\sum_{i\in f^{-1}(g^{-1}(\ell))}a_i\ \mathrm{mod}\ 2\right)_{\ell=1,...,k}\\
    &= \left(\sum_{j\in g^{-1}(\ell)}\left(\sum_{i\in f^{-1}(j)}a_i\ \mathrm{mod}\ 2\right)\ \mathrm{mod}\ 2\right)_{\ell=1,...,k}\\
    &= \tilde{\A}g\left(\left(\sum_{i\in f^{-1}(j)}a_i\ \mathrm{mod}\ 2\right)_{j=1,...,m}\right)\\
    &= \tilde{\A}g\left(\tilde{\A}f((a_i)_{i=1,...,k})\right)
\end{align}
(Here, the well-defineness of the function heavily relies on the assumption that if $f^{-1}(j)=\emptyset$, the corresponding sum is $0$). In particular, $\tilde{\A}(g\circ f) = \tilde{\A}g\circ \tilde{\A}f$, so $\tilde{\A}$ is a functor (or an $\F_G$-algebra), and since each $\tilde{\A}(\mathbf{n}^\alpha):= \A^{\mathbf{n}^\alpha}$, it's in fact an $\F_{G,\RR}$-algebra.

Yet, since $\A$ is discrete, with $H_0,H_1$ being distinct functors on $\A$, we see that not all the $G$-actions on $\A$ are naturally isomorphic to $\Id_\A$, violating the property described in previous proposition. Hence, $\A$ cannot be a $\cP_G$-pseudoalgebra, or even a $\cP_G$-pseudoalgebra due to the same reasoning.
\end{con}

\hfil

This shows that the functor $\RR_G:\cP_G$-$\PsAlg\rightarrow \F_G$-$\PsAlg$ is not essentially surjective, hence cannot be an equivalence of categories. Same statement holds true for $\cP_G$-$\PsAlg$.
